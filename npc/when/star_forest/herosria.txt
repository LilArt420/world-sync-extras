// 1 when the player's guild holds a castle, else 0.
function	script	F_WSF_HasCastle	{
	.@g = getcharid(2);
	if (!.@g)
		return 0;
	setarray .@c$,
		"prtg_cas01","prtg_cas02","prtg_cas03","prtg_cas04","prtg_cas05",
		"payg_cas01","payg_cas02","payg_cas03","payg_cas04","payg_cas05",
		"gefg_cas01","gefg_cas02","gefg_cas03","gefg_cas04","gefg_cas05",
		"aldeg_cas01","aldeg_cas02","aldeg_cas03","aldeg_cas04","aldeg_cas05",
		"arug_cas01","arug_cas02","arug_cas03","arug_cas04","arug_cas05",
		"schg_cas01","schg_cas02","schg_cas03","schg_cas04","schg_cas05",
		"te_aldecas1","te_aldecas2","te_aldecas3","te_aldecas4","te_aldecas5",
		"te_prtcas01","te_prtcas02","te_prtcas03","te_prtcas04","te_prtcas05";
	for (.@i = 0; .@i < getarraysize(.@c$); .@i++)
		if (getcastledata(.@c$[.@i], CD_GUILD_ID) == .@g)
			return 1;
	return 0;
}

// Turns <amount> of each input into one output, as many times as the
// player can and wants.
// callfunc "F_WSF_Craft", "<npc name>", <output id>, <zeny per craft>, <in id>, <amount>{, <in id>, <amount>...};
function	script	F_WSF_Craft	{
	.@npc$ = getarg(0);
	.@out = getarg(1);
	.@zeny = getarg(2);
	.@n = 0;
	for (.@a = 3; .@a < getargcount(); .@a += 2) {
		.@in[.@n] = getarg(.@a);
		.@amt[.@n] = getarg(.@a + 1);
		.@n++;
	}
	.@max = 999;
	for (.@i = 0; .@i < .@n; .@i++)
		.@max = min(.@max, countitem(.@in[.@i]) / .@amt[.@i]);
	if (.@zeny)
		.@max = min(.@max, Zeny / .@zeny);
	mes "[" + .@npc$ + "]";
	mes "One " + getitemname(.@out) + " takes:";
	for (.@i = 0; .@i < .@n; .@i++)
		mes "- " + .@amt[.@i] + " " + getitemname(.@in[.@i]) + " (you have " + countitem(.@in[.@i]) + ")";
	if (.@zeny)
		mes "- " + callfunc("F_InsertComma", .@zeny) + " zeny";
	if (.@max < 1) {
		mes "You don't have enough for even one.";
		close;
	}
	next;
	mes "[" + .@npc$ + "]";
	mes "You can make up to " + .@max + ". How many?";
	next;
	if (input(.@k, 0, .@max) != 0 || .@k == 0) {
		mes "[" + .@npc$ + "]";
		mes "Come back when you've decided.";
		close;
	}
	if (!checkweight(.@out, .@k)) {
		mes "[" + .@npc$ + "]";
		mes "You can't carry that much. Make some room first.";
		close;
	}
	for (.@i = 0; .@i < .@n; .@i++)
		delitem .@in[.@i], .@amt[.@i] * .@k;
	if (.@zeny)
		Zeny -= .@zeny * .@k;
	getitem .@out, .@k;
	specialeffect EF_REFINEOK;
	mes "[" + .@npc$ + "]";
	mes "Here you are: " + .@k + " " + getitemname(.@out) + ".";
	close;
}

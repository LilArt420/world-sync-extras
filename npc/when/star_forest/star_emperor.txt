// Who can take the job change test.
function	script	F_WSF_SeCan	{
	if (Class != JOB_STAR_GLADIATOR && Class != JOB_STAR_GLADIATOR2)
		return 0;
	if (BaseLevel < 99 || JobLevel < 50)
		return 1;
	return 2;
}

// The last step: the chosen master makes the player a Star Emperor.
// callfunc "F_WSF_SeChoose", <1 Ilgwang | 2 Chosung | 3 Eunha>;
function	script	F_WSF_SeChoose	{
	.@m = getarg(0);
	setarray .@name$, "", "Ilgwang", "Chosung", "Eunha";
	setarray .@road$, "", "the sun", "the moon", "the stars";
	.@n$ = "[" + .@name$[.@m] + "]";
	mes .@n$;
	switch (.@m) {
	case 1:
		mes "Master Chosi told me. You're choosing your teacher today.";
		break;
	case 2:
		mes "The master sent you back to choose. Hmph. Well?";
		break;
	default:
		mes "Oh, it's choosing day? I thought you'd pick Ilgwang. Everyone picks Ilgwang.";
		break;
	}
	next;
	if (select("I want to grow with you.:I'm still thinking.") == 2) {
		mes .@n$;
		mes "Then think. A choice like this should be made once.";
		close;
	}
	if (Class != JOB_STAR_GLADIATOR && Class != JOB_STAR_GLADIATOR2) {
		mes .@n$;
		mes "Wait. You're not a TaeKwon Master any more. Only a TaeKwon Master can become a Star Emperor.";
		close;
	}
	if (SkillPoint > 0) {
		mes .@n$;
		mes "Before that: you still have skill points left. Learn your skills first, then come back.";
		close;
	}
	mes .@n$;
	switch (.@m) {
	case 1:
		mes "Then we'll walk the road of " + .@road$[.@m] + " together. Slow, steady, and every day.";
		break;
	case 2:
		mes "Good. The road of " + .@road$[.@m] + " is hard. I won't go easy on you.";
		break;
	default:
		mes "Me? Really? Ha... then the road of " + .@road$[.@m] + " it is. We'll take it at night, when it's quiet.";
		break;
	}
	next;
	mes .@n$;
	mes "Close your eyes. Feel the sun, the moon and the stars above you.";
	close2;
	specialeffect2 EF_ANGEL;
	specialeffect2 EF_SPHERE;
	sleep2 1000;
	callfunc "F_WSF_Step", "wsf_se", 140, "11585", "";
	wsf_se_m = .@m;
	jobchange JOB_STAR_EMPEROR;
	specialeffect2 EF_JOBLVUP;
	announce strcharinfo(0) + " has become a Star Emperor in the Meteor Forest!", BC_MAP, 0xFFFF00;
	end;
}

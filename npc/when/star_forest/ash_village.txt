// callfunc "F_WSF_Sign", "<place>", "<map>", <x>, <y>;
function	script	F_WSF_Sign	{
	mes "^4d4dffA branch of the World Tree, its leaves pointing toward " + getarg(0) + ".^000000";
	next;
	if (select("Follow the branch:Stay") == 2)
		close;
	close2;
	warp getarg(1), getarg(2), getarg(3);
	end;
}

// Hot Beer More: the Tangled Magic armor series for Root Gold Coins,
// after the main story.
ygg_fruit,168,124,4	script	Hotbeer More#wsf	813,{
	mes "[Hotbeer More]";
	mes "Armor woven from the World Tree's own roots. Keeps the other worlds' magic off your skin.";
	if (!callfunc("F_WSF_Ch1Done")) {
		mes "But I only sell it to people the village trusts. Help Ash Village through its troubles first.";
		close;
	}
	mes "Each piece costs ^0000FF100 Root Gold Coins^000000.";
	next;
	setarray .@it, 450440, 450441, 480545, 480546, 470333, 470334;
	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@it); .@i++)
		.@menu$ += getitemname(.@it[.@i]) + ":";
	.@menu$ += "Never mind";
	.@c = select(.@menu$) - 1;
	if (.@c >= getarraysize(.@it))
		close;
	if (countitem(1001972) < 100) {
		mes "[Hotbeer More]";
		mes "That's 100 Root Gold Coins. You have " + countitem(1001972) + ".";
		close;
	}
	if (!checkweight(.@it[.@c], 1)) {
		mes "[Hotbeer More]";
		mes "You can't carry it. Make some room first.";
		close;
	}
	delitem 1001972, 100;	// Ch1_Root_Coin
	getitem .@it[.@c], 1;
	mes "[Hotbeer More]";
	mes "There you go. Wear it well.";
	close;
}

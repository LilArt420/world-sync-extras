	next;

		close;

// One of Hotbeer More's exchanges: pick an item, pay up to four materials.
// callfunc "F_WSF_MoreTrade", <item array>, <mat1>, <n1>, <mat2>, <n2>, <mat3>, <n3>, <mat4>, <n4>;
function	script	F_WSF_MoreTrade	{
	.@cnt = getarraysize(getarg(0));
	.@menu$ = "";
	for (.@i = 0; .@i < .@cnt; .@i++)
		.@menu$ += getitemname(getelementofarray(getarg(0), .@i)) + ":";
	.@menu$ += "Never mind";
	.@c = select(.@menu$) - 1;
	if (.@c >= .@cnt)
		close;
	.@item = getelementofarray(getarg(0), .@c);
	for (.@m = 0; .@m < 4; .@m++) {
		.@mat[.@m] = getarg(1 + .@m * 2);
		.@amt[.@m] = getarg(2 + .@m * 2);
	}
	mes "[Hotbeer More]";
	mes getitemname(.@item) + "? That takes:";
	for (.@m = 0; .@m < 4; .@m++)
		if (.@mat[.@m])
			mes "- " + .@amt[.@m] + " " + getitemname(.@mat[.@m]) + " (you have " + countitem(.@mat[.@m]) + ")";
	next;
	if (select("Make the trade:Never mind") == 2)
		close;
	for (.@m = 0; .@m < 4; .@m++) {
		if (.@mat[.@m] && countitem(.@mat[.@m]) < .@amt[.@m]) {
			mes "[Hotbeer More]";
			mes "You're short of " + getitemname(.@mat[.@m]) + ".";
			close;
		}
	}
	if (!checkweight(.@item, 1)) {
		mes "[Hotbeer More]";
		mes "You can't carry it. Make some room first.";
		close;
	}
	for (.@m = 0; .@m < 4; .@m++)
		if (.@mat[.@m])
			delitem .@mat[.@m], .@amt[.@m];
	getitem .@item, 1;
	mes "[Hotbeer More]";
	mes "There you go. Wear it well.";
	close;
}

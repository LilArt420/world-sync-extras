//= when it has handled the talk.

// Lets the Destroyed Geffen part speak through an Ash Village NPC.
// callfunc("F_WSF_Hook", "<npc key>") -> 1 when handled
function	script	F_WSF_Hook	{
	if (!$@wsv_ch1)
		return 0;
	return callfunc("F_WSV_AshHook", getarg(0));
}

// callfunc "F_WSF_Sign", "<place>", "<map>", <x>, <y>;
function	script	F_WSF_Sign	{
	mes "^4d4dffA branch of the World Tree, its leaves pointing toward " + getarg(0) + ".^000000";
	next;
	if (select("Follow the branch:Stay") == 2)
		close;
	close2;
	warp getarg(1), getarg(2), getarg(3);
	end;
}

// ---------------------------------------------------- Requests
// getarg(0): grade 1-3. Returns a random request quest id of that grade.
function	script	F_IQ_SSPick	{
	switch (getarg(0)) {
	case 1: setarray .@q, 8324, 8326, 8328, 8330, 8332, 8336, 8338, 8340; break;
	case 2: setarray .@q, 8334, 8342; break;
	case 3: setarray .@q, 8344, 8346, 8348, 8350, 8352; break;
	}
	return .@q[rand(getarraysize(.@q))];
}

// getarg(0): request quest id. Returns its grade.
function	script	F_IQ_SSGrade	{
	.@q = getarg(0);
	if (.@q == 8334 || .@q == 8342)
		return 2;
	if (.@q >= 8344)
		return 3;
	return 1;
}

function	script	F_IQ_SSGradeName	{
	setarray .@n$[1], "Beginner", "Regular", "Master";
	return .@n$[getarg(0)];
}

// getarg(0): text, getarg(1): max length. Mail titles over 40 characters are refused.
function	script	F_IQ_SSCut	{
	if (getstrlen(getarg(0)) <= getarg(1))
		return getarg(0);
	return substr(getarg(0), 0, getarg(1) - 1);
}

// getarg(0): item id, getarg(1): units to delete (0 or none: only count them).
// Only plain stacks count: tradable, not equipped, bound, rented, refined, carded,
// graded or with random options. delitem would take any copy, so a bound or rented
// one could leave the inventory while the other side gets a plain new one.
// Returns the units held, or when deleting, 1 on success.
function	script	F_IQ_SSPlain	{
	.@id = getarg(0);
	.@del = getarg(1, 0);
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@id || @inventorylist_equip[.@i] || @inventorylist_bound[.@i] || @inventorylist_expire[.@i]
		|| !@inventorylist_tradable[.@i] || !@inventorylist_identify[.@i] || @inventorylist_refine[.@i] || @inventorylist_attribute[.@i]
		|| @inventorylist_card1[.@i] || @inventorylist_card2[.@i] || @inventorylist_card3[.@i] || @inventorylist_card4[.@i]
		|| @inventorylist_enchantgrade[.@i] || @inventorylist_option_id1[.@i])
			continue;
		.@idx[.@n] = @inventorylist_idx[.@i];
		.@amt[.@n] = @inventorylist_amount[.@i];
		.@have += .@amt[.@n];
		.@n++;
	}
	if (!.@del)
		return .@have;
	if (.@del > .@have)
		return 0;
	for (.@j = 0; .@j < .@n && .@del > 0; .@j++) {
		.@d = min(.@del, .@amt[.@j]);
		if (!delitemidx(.@idx[.@j], .@d))
			return 0;
		.@del -= .@d;
	}
	return 1;
}

// getarg(0): slot. Closes the stall: mails back what's left, removes the NPC.
function	script	F_IQ_SSClose	{
	.@s = getarg(0);
	.@cid = $iqss_owner[.@s];
	if (!.@cid)
		return;
	.@p$ = "$iqss" + .@s + "_";
	if ($iqss_kind[.@s] == 1) {
		.@n = 0;
		for (.@k = 0; .@k < 5; .@k++) {
			if (getd(.@p$ + "amt[" + .@k + "]") > 0) {
				.@mid[.@n] = getd(.@p$ + "id[" + .@k + "]");
				.@mamt[.@n] = getd(.@p$ + "amt[" + .@k + "]");
				.@mref[.@n] = getd(.@p$ + "ref[" + .@k + "]");
				.@mbnd[.@n] = 0;
				.@mc0[.@n] = getd(.@p$ + "c0[" + .@k + "]");
				.@mc1[.@n] = getd(.@p$ + "c1[" + .@k + "]");
				.@mc2[.@n] = getd(.@p$ + "c2[" + .@k + "]");
				.@mc3[.@n] = getd(.@p$ + "c3[" + .@k + "]");
				.@n++;
			}
		}
		if (.@n)
			mail .@cid, "Street Stall Staffer", "Your stall has closed", "Your selling staffer has gone home. Here is what was left unsold.", 0, .@mid, .@mamt, .@mref, .@mbnd, .@mc0, .@mc1, .@mc2, .@mc3;
	} else if ($iqss_esc[.@s] > 0)
		mail .@cid, "Street Stall Staffer", "Your stall has closed", "Your buying staffer has gone home. Here is the zeny that wasn't spent.", $iqss_esc[.@s];
	if ($iqss_npc$[.@s] != "")
		unloadnpc $iqss_npc$[.@s];
	for (.@k = 0; .@k < 5; .@k++) {
		setd .@p$ + "id[" + .@k + "]", 0;
		setd .@p$ + "amt[" + .@k + "]", 0;
		setd .@p$ + "price[" + .@k + "]", 0;
		setd .@p$ + "ref[" + .@k + "]", 0;
		setd .@p$ + "c0[" + .@k + "]", 0;
		setd .@p$ + "c1[" + .@k + "]", 0;
		setd .@p$ + "c2[" + .@k + "]", 0;
		setd .@p$ + "c3[" + .@k + "]", 0;
	}
	$iqss_owner[.@s] = 0;
	$iqss_owner$[.@s] = "";
	$iqss_kind[.@s] = 0;
	$iqss_end[.@s] = 0;
	$iqss_esc[.@s] = 0;
	$iqss_title$[.@s] = "";
	$iqss_npc$[.@s] = "";
	return;
}

		if (getd(.@p$ + "amt[" + .@k + "]") > 0)
			close;
	$iqss_end[.@s] = 0;
	close;
}

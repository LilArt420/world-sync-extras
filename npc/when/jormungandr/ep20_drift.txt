// Every monster of a wave runs this when it dies; the last one moves the
// instance on. The killer is attached when there is one (for the marks).
OnWave:
	.@m$ = instance_mapname("1@twsd");
	if (mobcount(.@m$, instance_npcname("#wsj_sd") + "::OnWave") > 0)
		end;
	switch ('sd) {
	case 1:
		'sd = 2;
		disablenpc instance_npcname("Rgan Bishop#wsj_sd1");
		mapannounce .@m$, "Rgan Bishop: The shamans! Protect the ritual!", bc_map, "0xFF4444";
		.@x = 319; .@y = 100;
		break;
	case 3:
		'sd = 4;
		enablenpc instance_npcname("Floe Crossing#wsj_sd1");
		mapannounce .@m$, "The ritual hole freezes over. A path of shimmering ice opens across the water.", bc_map, "0x88CCFF";
		.@x = 294; .@y = 115;
		break;
	case 5:
		'sd = 6;
		enablenpc instance_npcname("Floe Crossing#wsj_sd2");
		mapannounce .@m$, "The last shaman on the floe falls. Another crossing shimmers to the west.", bc_map, "0x88CCFF";
		.@x = 220; .@y = 154;
		break;
	case 8:
		'sd = 9;
		mapannounce .@m$, "Rgan Bishop: Guardians! To me!", bc_map, "0xFF4444";
		callfunc "F_WSJ_SdWave", 21991, 1, 163, 194;
		callfunc "F_WSJ_SdWave", 21992, 1, 163, 194;
		end;
	case 9:
		'sd = 10;
		mapannounce .@m$, "Rgan Bishop: Shamans, rise! All of you!", bc_map, "0xFF4444";
		callfunc "F_WSJ_SdWave", 21993, 2, 163, 194;
		end;
	case 10:
		'sd = 11;
		disablenpc instance_npcname("Rgan Bishop#wsj_sd2");
		enablenpc instance_npcname("Lehar#wsj_sd1");
		mapannounce .@m$, "The bishop dives into a crack in the ice and is gone. Someone is calling your name from the north.", bc_map, "0x88CCFF";
		.@x = 162; .@y = 245;
		break;
	case 13:
		'sd = 14;
		enablenpc instance_npcname("Sarekgand#wsj_sd");
		enablenpc instance_npcname("Bagot#wsj_sd");
		mapannounce .@m$, "Voices carry across the water, from the last floe.", bc_map, "0x88CCFF";
		.@x = 48; .@y = 326;
		break;
	case 15:
		'sd = 16;
		mapannounce .@m$, "Sarekgand: ...Impressive. Annoying, but impressive.", bc_map, "0xFF4444";
		.@x = 48; .@y = 326;
		break;
	case 17:
		'sd = 18;
		mapannounce .@m$, "The last ritual circle breaks. Somebody sighs, loudly.", bc_map, "0x88CCFF";
		.@x = 46; .@y = 328;
		break;
	default:
		end;
	}
	if (playerattached())
		callfunc "F_WSJ_CnMark", .@x, .@y;
	end;
}

// A wave of the cult's guards around a spot, counted by #wsj_sd::OnWave.
// callfunc "F_WSJ_SdWave", <mob id>, <amount>, <x>, <y>;
function	script	F_WSJ_SdWave	{
	.@x = getarg(2);
	.@y = getarg(3);
	areamonster instance_mapname("1@twsd"), .@x - 3, .@y - 3, .@x + 3, .@y + 3, "--ja--", getarg(0), getarg(1), instance_npcname("#wsj_sd") + "::OnWave";
	return;
}

// Marks the next place to go on the minimap (one mark at a time).
function	script	F_WSJ_CnMark	{
	viewpoint 2, 1, 1, 1, 0xFFFFFF;
	if (getarg(0) > 0)
		viewpoint 1, getarg(0), getarg(1), 1, 0x88CCFF;
	return;
}

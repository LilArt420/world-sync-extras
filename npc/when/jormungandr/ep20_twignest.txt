1@twig	mapflag	nosave	SavePoint
1@twig	mapflag	noteleport
1@twig	mapflag	nomemo
1@twig	mapflag	nowarpto
1@twig	mapflag	nobranch
1@twig	mapflag	monster_noteleport

1@twig,1,1,0	script	#wsj_tn	HIDDEN_WARP_NPC,{
	end;
OnInstanceInit:
	'tn = 0;
	disablenpc instance_npcname("Lehar#wsj_tn2");
	disablenpc instance_npcname("Lehar#wsj_tn3");
	disablenpc instance_npcname("Leon#wsj_tn");
	disablenpc instance_npcname("Aurelie#wsj_tn");
	disablenpc instance_npcname("White Cat#wsj_tn");
	end;

// The last guardian of a pack moves the instance on.
OnWave:
	.@m$ = instance_mapname("1@twig");
	if (mobcount(.@m$, instance_npcname("#wsj_tn") + "::OnWave") > 0)
		end;
	if ('tn == 1) {
		'tn = 2;
		mapannounce .@m$, "Lehar: Over here! Quick, before more of them grow!", bc_map, "0x88CCFF";
		.@x = 160; .@y = 88;
	} else if ('tn == 3) {
		'tn = 4;
		mapannounce .@m$, "Lehar: This way! I can hear someone up ahead!", bc_map, "0x88CCFF";
		.@x = 157; .@y = 140;
	} else
		end;
	if (playerattached())
		callfunc "F_WSJ_CnMark", .@x, .@y;
	end;
}

// A pack of the nest's guardians around a spot, counted by #wsj_tn::OnWave.
// callfunc "F_WSJ_TnWave", <x>, <y>;
function	script	F_WSJ_TnWave	{
	.@x = getarg(0);
	.@y = getarg(1);
	.@m$ = instance_mapname("1@twig");
	.@e$ = instance_npcname("#wsj_tn") + "::OnWave";
	areamonster .@m$, .@x - 3, .@y - 3, .@x + 3, .@y + 3, "--ja--", 21991, 2, .@e$;
	areamonster .@m$, .@x - 3, .@y - 3, .@x + 3, .@y + 3, "--ja--", 21992, 1, .@e$;
	return;
}

// Moves the story on: completes the quest(s) in "done", starts "next".
// callfunc "F_WSJ_Koko", <new step>, "<done ids>", "<next ids>";
function	script	F_WSJ_Koko	{
	explode .@d$, getarg(1), ",";
	.@n = getarraysize(.@d$);
	for (.@i = 0; .@i < .@n; .@i++)
		if (.@d$[.@i] != "" && isbegin_quest(atoi(.@d$[.@i])) == 1)
			completequest atoi(.@d$[.@i]);
	explode .@x$, getarg(2), ",";
	.@n = getarraysize(.@x$);
	for (.@i = 0; .@i < .@n; .@i++)
		if (.@x$[.@i] != "" && isbegin_quest(atoi(.@x$[.@i])) == 0)
			setquest atoi(.@x$[.@i]);
	wsj_koko = getarg(0);
	return;
}

	next;

	case 2:
		if (.@d == 2) {
			mes "[Lalaha]";
			mes "No more Kokos in the Sticky Sea today. I counted.";
			close;
		}
		if (.@d == 0) {
			mes "[Lalaha]";
			mes "Some Kokos have gone down into the Sticky Sea again. Five of them, I think. Could you find them and give them a good scolding? Then send them home.";
			wsj_kscold = 0;
			setquest 11951;
			close;
		}
		if (wsj_kscold != 31) {
			mes "[Lalaha]";
			mes "There are still Kokos down there. Five in all. The Koko by the hole on the beach will show you in.";
			close;
		}
		mes "[Lalaha]";
		mes "All five back, sticky and sorry. Thank you.";
		wsj_kscold = 0;
		callfunc "F_WSJ_DailyDone", 11951, 11952, 15000000, 1;
		close;
	case 3:
		callfunc "F_WSJ_KokoWind", .@a, 11953, 11954, "Ashen Icewinds", "the eastern part of the Ancient Ice Canyon";
		end;
	case 4:
		callfunc "F_WSJ_KokoWind", .@b, 11955, 11956, "Icewinds", "the western part of the Ancient Ice Canyon";
		end;
	}
	close;
}

function	script	F_WSJ_KokoWind	{
	mes "[Lalaha]";

// A transport: "<npc>", "<line>", "<choice>", "<map>", x, y
function	script	F_WSJ_Ride	{
	mes "[" + getarg(0) + "]";
	mes getarg(1);
	next;
	if (select(getarg(2) + ":Not now.") == 2)
		close;
	close2;
	warp getarg(3), getarg(4), getarg(5);
	end;
}

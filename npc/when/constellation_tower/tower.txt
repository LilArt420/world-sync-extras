//===== World Sync Extras: Constellation Tower block ========
//= The Constellation Tower (kRO 2023): the Endless Tower, taken over
//= and turned upside down by the constellation Betelgeuse. rAthena has
//= the instance ("Constellation Tower", 1@ch_t, 2@ch_t, 3@ch_t) but no
//= script and no monsters. kRO's own walkthrough isn't published; this
//= follows the client's quest log (12513-12519), its tip box (level 240,
//= every 3 days, 60 minutes, Oscar and OSC0007 on Misty Island) and
//= RockMMO's guide to its version of the tower.
//=
//= - Floors: 1@ch_t is 25 identical rooms. Floor N is room (N-1) % 25,
//=   so the climb goes round the rooms again after floor 25, with the
//=   monsters 50% stronger each time round. Each floor: three of the old
//=   tower's monsters (every 5th floor: one of its bosses and two more).
//=   Clear the floor and its stairway opens; everyone still in the room
//=   is carried up with the first who takes it. Each cleared floor gives
//=   everyone in the tower Meteorite Powder.
//= - Floor 25: end the climb against Naght Sieger, the tower's cast-out
//=   owner (2@ch_t), or keep climbing. Floor 50 and 70: Betelgeuse
//=   (3@ch_t), harder from floor 70; his fountain makes him harder still
//=   for Meteorite Powder, and his chest pays more.
//= - Oscar starts a new climb on floor 1, or higher for Meteorite Powder
//=   (40 per 5 floors, up to floor 25).
//= Rewards, floor-skip prices, the fountain and the exchange are our
//= estimates; monster stats and drops are Divine Pride's.
//=
//= Char vars: wsct_iid (the tower instance the character entered).
//============================================================

		mes "[Oscar]";
		mes "There's a new owner in there. A constellation, like me: ^0000FFBetelgeuse^000000. He calls it the Constellation Tower now, and he's not a kind landlord.";
		mes "I'd like to know what happened to the old owner, too. Will you climb it?";
		setquest 12513;
		next;
	}
	mes "[Oscar]";
	mes "The Constellation Tower. Base level 240 and up, in a party. Once inside, you have an hour; after you leave, the tower won't let you back in for three days.";
	next;
	switch (select("Enter the tower.:About the tower.:Leave.")) {
	case 2:
		mes "[Oscar]";
		mes "Each floor is full of the old tower's monsters, stronger than they ever were. Clear a floor and the stairway opens.";
		mes "After the 25th floor you can face the tower's old owner and leave, or climb on. Betelgeuse waits at the 50th floor, and harder at the 70th.";
		next;

		mes "Not alone. Form a party, even a party of one, and come back.";

// Rooms in climbing order: their lower-left corners. The entry is at
// (+44,+9) in each room, its stairway at (+20,+42).
function	script	F_WSCT_Room	{
	setarray .@x, 4,88,176,262,346,4,88,176,262,346,4,88,176,262,346,4,88,176,262,346,4,88,176,262,346;
	setarray .@y, 174,174,174,174,174,90,90,90,90,90,4,4,4,4,4,346,346,346,346,346,262,262,262,262,262;
	.@r = (getarg(0) - 1) % 25;
	set getarg(1), .@x[.@r];
	set getarg(2), .@y[.@r];
	return .@r;
}

// 'floor: the floor being fought (0 before anyone arrives); 'cleared: the
// highest floor cleared; 'route: 0 climbing, 1 Naght Sieger, 2 Betelgeuse.
1@ch_t,1,1,0	script	#wsct	-1,{
	end;
OnInstanceInit:
	'floor = 0;
	'cleared = 0;
	'route = 0;
	for (.@i = 1; .@i <= 25; .@i++)
		disablenpc instance_npcname("Stairway#wsct" + .@i);
	end;

// Starts floor getarg(0): its monsters, and moves everyone from the
// previous floor's room.
S_Floor:
	.@n = getarg(0);
	.@m$ = instance_mapname("1@ch_t");
	if ('floor > 0) {
		disablenpc instance_npcname("Stairway#wsct" + (callfunc("F_WSCT_Room", 'floor, .@px, .@py) + 1));
	}
	'floor = .@n;
	callfunc "F_WSCT_Room", .@n, .@x, .@y;
	if (.@px || .@py)
		areawarp .@m$, .@px, .@py, .@px + 49, .@py + 49, .@m$, .@x + 44, .@y + 9;
	// Normal monsters 20998-21047 (the four boss kinds aside).
	setarray .@boss, 21002, 21019, 21031, 21044;
	.@lap = (.@n - 1) / 25;
	.@count = 3;
	'left = 3;
	.@e$ = instance_npcname("#wsct") + "::OnMob";
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@i == 0 && .@n % 5 == 0)
			.@mob = .@boss[rand(4)];
		else {
			do {
				.@mob = rand(20998, 21047);
			} while (inarray(.@boss, .@mob) != -1);
		}
		areamonster .@m$, .@x + 4, .@y + 4, .@x + 45, .@y + 45, "--ja--", .@mob, 1, .@e$;
		if (.@lap > 0) {
			getunitdata $@mobid[0], .@d;
			.@hp = .@d[UMOB_MAXHP] / 2 * (2 + .@lap);
			setunitdata $@mobid[0], UMOB_MAXHP, .@hp;
			setunitdata $@mobid[0], UMOB_HP, .@hp;
			setunitdata $@mobid[0], UMOB_ATKMIN, .@d[UMOB_ATKMIN] / 2 * (2 + .@lap);
			setunitdata $@mobid[0], UMOB_ATKMAX, .@d[UMOB_ATKMAX] / 2 * (2 + .@lap);
		}
	}
	mapannounce .@m$, "Constellation Tower: floor " + .@n + (.@n % 5 == 0 ? ". One of the old tower's masters guards it." : "."), bc_map, "0x88CCFF";
	return;

OnStart:
	if ('floor > 0)
		end;
	callsub S_Floor, max(1, $@wsct_start[instance_id()]);
	$@wsct_start[instance_id()] = 0;
	end;

OnClimb:
	if ('cleared != 'floor || 'route)
		end;
	callsub S_Floor, 'floor + 1;
	end;

OnMob:
	'left--;
	if ('left > 0)
		end;
	'cleared = 'floor;
	.@m$ = instance_mapname("1@ch_t");
	// Everyone in the tower gets Meteorite Powder: 2 a floor, 5 on every 5th.
	.@p = ('floor % 5 == 0) ? 5 : 2;
	.@n = getmapunits(BL_PC, .@m$, .@pc);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (attachrid(.@pc[.@i])) {
			getitem 1000372, .@p;	// Meteorite_Dust
			detachrid;
		}
	}
	enablenpc instance_npcname("Stairway#wsct" + (callfunc("F_WSCT_Room", 'floor, .@x, .@y) + 1));
	if ('floor == 25)
		mapannounce .@m$, "Floor 25 is clear. The stairway leads on, or down to the hall where the tower's old owner was thrown.", bc_map, "0x88CCFF";
	else if ('floor == 50 || 'floor == 70)
		mapannounce .@m$, "Floor " + 'floor + " is clear. Above the stairway, something vast is waiting: the tower's new owner.", bc_map, "0xFF4444";
	else
		mapannounce .@m$, "Floor " + 'floor + " is clear. The stairway is open.", bc_map, "0x88CCFF";
	end;
}

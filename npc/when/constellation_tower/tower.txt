//===== World Sync Extras: Constellation Tower block ========
//= The Constellation Tower (kRO 2023): the Endless Tower, taken over
//= and turned upside down by the constellation Betelgeuse. rAthena has
//= the instance ("Constellation Tower", 1@ch_t, 2@ch_t, 3@ch_t) but no
//= script. This follows iROWiki's Constellation Tower page (floors,
//= bosses, skips, the fountain's stars, the exchange), with the monster
//= counts of Midgard Tales' guide and the boss fights' habits of the
//= kRO-based guides (Midgard Tales, Shining Moon, RockMMO):
//=
//= - Floors: 1@ch_t is 25 identical rooms; floor N is room (N-1) % 25.
//=   Each floor holds 2 to 14 of the old tower's monsters, a little
//=   stronger every floor; clear them and the stairway opens. Every 5th
//=   floor also holds one of the old tower's bosses (21325-21359), who
//=   doesn't have to be beaten to go on. The blue brazier at the start of
//=   floors 1, 6, 11... skips 4 floors for Meteorite Powder (50, then 25
//=   more each time).
//= - Floor 25: fight Naght Sieger, the tower's cast-out owner (2@ch_t),
//=   or climb on. Floor 50: fight Betelgeuse (3@ch_t), stars 1-3, or climb
//=   on to floor 75 and fight him at 3-5 stars.
//= - Betelgeuse's fountain: five stats (Health, Attack, Recovery,
//=   Defense, Traps), each 0 (sealed) to 5 stars. Reroll all for 20
//=   Meteorite Powder, max one for 150, seal one for 300. Their sum is his
//=   strength (all 1 star = 200%, all 3 = 600%, all 5 = 1000%); his chest
//=   pays one share at 200%, and one more for every 100% above that.
//= - The monsters drop the Meteorite Powder (0.5%, Divine Pride); the
//=   final bosses' chests pay the rest. OSC0007 trades it for the Faint
//=   Star and Nebula gear, OSC0008 enchants that gear once you've beaten
//=   one of the two owners.
//= Chest amounts beyond the souls are estimates.
//=
//= Char vars: wsct_iid (the tower instance the character entered).
//============================================================

// Rooms in climbing order: their lower-left corners. The entry is at
// (+44,+9) in each room, its stairway at (+20,+42), the blue brazier at
// (+46,+14).
function	script	F_WSCT_Room	{
	setarray .@x, 4,88,176,262,346,4,88,176,262,346,4,88,176,262,346,4,88,176,262,346,4,88,176,262,346;
	setarray .@y, 174,174,174,174,174,90,90,90,90,90,4,4,4,4,4,346,346,346,346,346,262,262,262,262,262;
	.@r = (getarg(0) - 1) % 25;
	set getarg(1), .@x[.@r];
	set getarg(2), .@y[.@r];
	return .@r;
}

// 'floor: the floor being fought (0 before anyone arrives); 'cleared: the
// highest floor cleared; 'route: 0 climbing, 1 Naght Sieger, 2 Betelgeuse;
// 'left: monsters still to beat; 'skips: braziers used; 'boss: the GID of
// the floor's boss.
1@ch_t,1,1,0	script	#wsct	-1,{
	end;
OnInstanceInit:
	'floor = 0;
	'cleared = 0;
	'route = 0;
	'skips = 0;
	'boss = 0;
	for (.@i = 1; .@i <= 25; .@i++)
		disablenpc instance_npcname("Stairway#wsct" + .@i);
	for (.@i = 1; .@i <= 5; .@i++)
		disablenpc instance_npcname("Blue Brazier#wsct" + .@i);
	end;

// Starts floor getarg(0): its monsters, and moves everyone from the
// previous floor's room.
S_Floor:
	.@n = getarg(0);
	.@m$ = instance_mapname("1@ch_t");
	if ('floor > 0) {
		.@r = callfunc("F_WSCT_Room", 'floor, .@px, .@py);
		disablenpc instance_npcname("Stairway#wsct" + (.@r + 1));
		if (.@r % 5 == 0)
			disablenpc instance_npcname("Blue Brazier#wsct" + (.@r / 5 + 1));
		killmonster .@m$, instance_npcname("#wsct") + "::OnMob";
	}
	// Cleared first: a boss removed here isn't a kill (OnBossKill).
	.@b = 'boss;
	'boss = 0;
	if (.@b && unitexists(.@b))
		unitkill .@b;
	'floor = .@n;
	.@r = callfunc("F_WSCT_Room", .@n, .@x, .@y);
	if (.@px || .@py)
		areawarp .@m$, .@px, .@py, .@px + 49, .@py + 49, .@m$, .@x + 44, .@y + 9;
	// 2 to 14 of the old tower's monsters (20998-21047), at 42% of Divine
	// Pride's strength on floor 1, 100% on floor 25, 220% on floor 75.
	.@count = rand(2, 14);
	.@hp_pct = 40 + 12 * .@n / 5;
	.@atk_pct = 60 + 8 * .@n / 5;
	'left = .@count;
	.@e$ = instance_npcname("#wsct") + "::OnMob";
	for (.@i = 0; .@i < .@count; .@i++) {
		areamonster .@m$, .@x + 4, .@y + 4, .@x + 45, .@y + 45, "--ja--", rand(20998, 21047), 1, .@e$;
		.@gid = $@mobid[0];
		getunitdata .@gid, .@d;
		.@hp = .@d[UMOB_MAXHP] / 100 * .@hp_pct;
		setunitdata .@gid, UMOB_MAXHP, .@hp;
		setunitdata .@gid, UMOB_HP, .@hp;
		setunitdata .@gid, UMOB_ATKMIN, .@d[UMOB_ATKMIN] * .@atk_pct / 100;
		setunitdata .@gid, UMOB_ATKMAX, .@d[UMOB_ATKMAX] * .@atk_pct / 100;
	}
	// Every 5th floor: one of the old tower's bosses, not needed to go on.
	if (.@n % 5 == 0) {
		areamonster .@m$, .@x + 15, .@y + 15, .@x + 35, .@y + 35, "--ja--", rand(21325, 21359), 1, instance_npcname("#wsct") + "::OnBossKill";
		'boss = $@mobid[0];
		getunitdata 'boss, .@d;
		.@b$ = getmonsterinfo(.@d[UMOB_CLASS], MOB_NAME);
		mapannounce .@m$, "Constellation Tower: floor " + .@n + ". " + .@count + " monsters, and " + .@b$ + ", one of the old tower's masters, guards it.", bc_map, "0x88CCFF";
	} else
		mapannounce .@m$, "Constellation Tower: floor " + .@n + ". " + .@count + " monsters.", bc_map, "0x88CCFF";
	// The brazier at the start of floors 1, 6, 11, ...
	if (.@n % 5 == 1 && .@n + 4 <= 75)
		enablenpc instance_npcname("Blue Brazier#wsct" + (.@r / 5 + 1));
	return;

OnStart:
	if ('floor > 0)
		end;
	callsub S_Floor, 1;
	end;

OnClimb:
	if ('cleared != 'floor || 'route)
		end;
	callsub S_Floor, 'floor + 1;
	end;

OnSkip:
	if ('cleared >= 'floor || 'route || 'floor % 5 != 1)
		end;
	'skips++;
	callsub S_Floor, 'floor + 4;
	end;

// A 5th-floor boss beaten: Meteorite Fragments for everyone in its room,
// 2 up to floor 20, 3 from 25, 4 from 50, 5 on 75 (an estimate: the
// official amount isn't published). No player is attached here.
OnBossKill:
	if ('boss == 0 || killedgid != 'boss)
		end;
	'boss = 0;
	.@m$ = instance_mapname("1@ch_t");
	callfunc "F_WSCT_Room", 'floor, .@x, .@y;
	.@amt = 2 + 'floor / 25;
	.@n = getmapunits(BL_PC, .@m$, .@pc);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (getmapxy(.@mm$, .@px, .@py, BL_PC, .@pc[.@i]) != 0)
			continue;
		if (.@px < .@x || .@px > .@x + 49 || .@py < .@y || .@py > .@y + 49)
			continue;
		if (attachrid(.@pc[.@i])) {
			getitem 1000373, .@amt;	// Meteorite_Fragment
			dispbottom "The old tower's master leaves " + .@amt + " Meteorite Fragments behind.";
			detachrid;
		}
	}
	mapannounce .@m$, "The floor's boss falls. Fragments of the stars scatter across the room.", bc_map, "0x88CCFF";
	end;

OnMob:
	'left--;
	if ('left > 0)
		end;
	'cleared = 'floor;
	.@m$ = instance_mapname("1@ch_t");
	.@r = callfunc("F_WSCT_Room", 'floor, .@x, .@y);
	enablenpc instance_npcname("Stairway#wsct" + (.@r + 1));
	if (.@r % 5 == 0)
		disablenpc instance_npcname("Blue Brazier#wsct" + (.@r / 5 + 1));
	if ('floor == 25)
		mapannounce .@m$, "Floor 25 is clear. The stairway leads on, or down to the hall where the tower's old owner was thrown.", bc_map, "0x88CCFF";
	else if ('floor == 50 || 'floor == 75)
		mapannounce .@m$, "Floor " + 'floor + " is clear. Above the stairway, something vast is waiting: the tower's new owner.", bc_map, "0xFF4444";
	else
		mapannounce .@m$, "Floor " + 'floor + " is clear. The stairway is open.", bc_map, "0x88CCFF";
	end;
}

// ------------------------------------------------------------
// Betelgeuse's throne (3@ch_t)
// ------------------------------------------------------------
// The ante-room (4-64, 82-101) holds the fountain (46,90) and the throne
// door (32,100); the throne room (126-185, 64-150).
// 'star[0..4]: Health, Attack, Recovery, Defense, Traps (0-5 stars);
// 'sealed: bit k set = stat k sealed at 0; 'cap: the most stars (3 after
// floor 50, 5 after floor 75); 'rollmin: the fewest a roll gives.

// What stat getarg(0) adds to Betelgeuse's strength at getarg(1) stars (%).
function	script	F_WSCT_Pct	{
	switch (getarg(0)) {
	case 0: setarray .@t, 0, 15, 30, 45, 60, 75; break;	// Health
	case 1: setarray .@t, 0, 20, 40, 60, 80, 100; break;	// Attack
	case 2: setarray .@t, 0, 30, 60, 90, 120, 150; break;	// Recovery
	case 3: setarray .@t, 0, 50, 100, 150, 200, 250; break;	// Defense
	case 4: setarray .@t, 0, 85, 170, 255, 340, 425; break;	// Traps
	}
	return .@t[getarg(1)];
}

3@ch_t,1,1,0	script	#wsct_throne	-1,{
	end;
OnInstanceInit:
	'throne = 0;
	'sealed = 0;
	disablenpc instance_npcname("Star Chest#wsct");
	end;
// Betelgeuse's strength (%) from the stars.
S_Diff:
	.@d = 0;
	for (.@k = 0; .@k < 5; .@k++)
		.@d += callfunc("F_WSCT_Pct", .@k, 'star[.@k]);
	return .@d;
OnRoll:
	'cap = ('last >= 75) ? 5 : 3;
	'rollmin = ('last >= 75) ? 3 : 1;
	for (.@k = 0; .@k < 5; .@k++)
		'star[.@k] = rand('rollmin, 'cap);
	end;
OnFight:
	if ('throne != 0)
		end;
	'throne = 1;
	.@m$ = instance_mapname("3@ch_t");
	.@diff = callsub(S_Diff);
	'diff = .@diff;
	.@h = callfunc("F_WSCT_Pct", 0, 'star[0]);
	.@a = callfunc("F_WSCT_Pct", 1, 'star[1]);
	.@r = callfunc("F_WSCT_Pct", 2, 'star[2]);
	.@f = callfunc("F_WSCT_Pct", 3, 'star[3]);
	.@t = callfunc("F_WSCT_Pct", 4, 'star[4]);
	monster .@m$, 156, 120, "Betelgeuse", 20994, 1, instance_npcname("#wsct_throne") + "::OnBoss";
	.@gid = $@mobid[0];
	'betel = .@gid;
	// 1,000,000,000 HP, +Health%; attack +Attack%; defense +Defense%.
	.@hp = 10000000 * (100 + .@h);
	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP, .@hp;
	getunitdata .@gid, .@d;
	setunitdata .@gid, UMOB_ATKMIN, .@d[UMOB_ATKMIN] * (100 + .@a) / 100;
	setunitdata .@gid, UMOB_ATKMAX, .@d[UMOB_ATKMAX] * (100 + .@a) / 100;
	setunitdata .@gid, UMOB_DEF, .@d[UMOB_DEF] * (100 + .@f) / 100;
	setunitdata .@gid, UMOB_MDEF, .@d[UMOB_MDEF] * (100 + .@f) / 100;
	mapannounce .@m$, "Betelgeuse: \"You climbed my tower to kneel at my throne? Then kneel.\" (strength " + .@diff + "%)", bc_map, "0xFF4444";
	// Every 5 seconds: his Dead Souls rise every 20 (two, six at most),
	// his traps every 15, and he recovers every 30.
	.@e$ = instance_npcname("#wsct_throne") + "::OnSoul";
	.@tick = 0;
	while ('throne == 1) {
		sleep 5000;
		if ('throne != 1 || !unitexists(.@gid))
			break;
		.@tick += 5;
		if (.@tick % 20 == 0) {
			.@alive = mobcount(.@m$, .@e$);
			if (.@alive < 6)
				areamonster .@m$, 140, 105, 172, 135, "Dead Soul", 20995, min(2, 6 - .@alive), .@e$;
		}
		if (.@r > 0 && .@tick % 30 == 0) {
			getunitdata .@gid, .@d;
			setunitdata .@gid, UMOB_HP, min(.@d[UMOB_MAXHP], .@d[UMOB_HP] + .@d[UMOB_MAXHP] / 1500 * .@r);
			unittalk .@gid, "Betelgeuse: The stars mend me.";
		}
		if (.@t > 0 && .@tick % 15 == 0) {
			// A trap snaps shut under each player two seconds after its
			// glow: step away. Damage: Traps% / 25 of max HP.
			deletearray .@pc;
			.@n = getmapunits(BL_PC, .@m$, .@pc);
			for (.@i = 0; .@i < .@n; .@i++) {
				if (attachrid(.@pc[.@i])) {
					getmapxy(.@mm$, .@px[.@i], .@py[.@i]);
					if (.@mm$ == .@m$ && .@px[.@i] >= 126)
						specialeffect2 EF_BEGINSPELL_N5;
					else
						.@px[.@i] = -10;
					detachrid;
				}
			}
			sleep 2000;
			for (.@i = 0; .@i < .@n; .@i++) {
				if (.@px[.@i] < 0 || !attachrid(.@pc[.@i]))
					continue;
				getmapxy(.@mm$, .@x, .@y);
				if (.@mm$ == .@m$ && Hp > 0 && distance(.@x, .@y, .@px[.@i], .@py[.@i]) <= 1) {
					specialeffect2 EF_FIREPILLARBOMB;
					percentheal -(.@t / 25), 0;
				}
				detachrid;
			}
		}
	}
	end;
OnSoul:
	end;
OnBoss:

// Enters (or creates, then enters) the party's memorial dungeon.
// Returns the instance id, or 0 after telling the player why not.
function	script	F_WSG_Enter	{
	.@md$ = getarg(0);
	if (instance_enter(.@md$) == IE_OK)
		return instance_id(IM_PARTY);
	.@id = instance_create(.@md$, IM_PARTY);
	if (.@id > 0 && instance_enter(.@md$) == IE_OK)
		return .@id;
	mes "[Dimensional Prison]";
	if (.@id == -3)
		mes "^ff0000Your party already has another memorial dungeon open. It must close before this gate will open.^000000";
	else
		mes "^ff0000The gate flickers and stays shut. Try again in a moment.^000000";
	close2;
	return 0;
}

// Common checks before either gate opens. Returns 1 if the player may go on.
function	script	F_WSG_GateCheck	{
	if (checkweight(1201,1) == 0) {
		mes "^ff0000It seems that there are too many types and weights of items. Please organize your inventory.^000000";
		close2;
		return 0;
	}
	if (BaseLevel < 250) {
		mes "[Dimensional Prison]";
		mes "^ff0000Only adventurers of base level 250 or higher can hold their own inside the prison.^000000";
		close2;
		return 0;
	}
	if (getcharid(1) < 1) {
		mes "^0000ffPlease organize a party of at least one person and proceed.^000000";
		close2;
		return 0;
	}
	return 1;
}

// The barrier level engraved on the Constellation's Protection the player
// wears (Constellation's Barrier (1)-(20) in its fourth slot), or 0 if they
// don't wear one.
function	script	F_WSG_Barrier	{
	if (getequipid(EQI_HEAD_LOW) != 420231)
		return 0;
	.@c = getequipcardid(EQI_HEAD_LOW, 3);
	if (.@c < 312453 || .@c > 312472)
		return 1;
	return .@c - 312452;
}

// Engraves a new barrier level on the worn Constellation's Protection.
function	script	F_WSG_SetBarrier	{
	.@lv = cap_value(getarg(0), 1, 20);
	if (getequipid(EQI_HEAD_LOW) != 420231)
		return;
	.@ref = getequiprefinerycnt(EQI_HEAD_LOW);
	.@c = getequipcardid(EQI_HEAD_LOW, 3);
	unequip EQI_HEAD_LOW;
	delitem2 420231,1,1,.@ref,0,0,0,0,.@c;
	getitem2 420231,1,1,.@ref,0,0,0,0,312452 + .@lv;	// MD_Hol_Protection
	equip 420231;
	return;
}

// Two mirrored halls: the prison (35-73, 30-68, centre 54,49) and the
// other world (333-371, 30-68).
//  - Traps: every 8 seconds, everyone standing far from the prison's
//    centre is struck (harder further out); each hit makes the criminal
//    stronger.
//  - Ten trap hits open a portal to the other world at the north wall.
//    Each pass through it makes the criminal heal faster. Killing the
//    three illusions there stuns the criminal for 20 seconds: no healing,
//    defenses broken, traps and power reset.
//  - Sanctuary: now and then the criminal makes itself immune where it
//    stands; pull it more than 4 cells away to break it.
//  - Barrier level ('level, from the protection of whoever opened the
//    hall): 400,000,000 HP, +100,000,000 per level over 1 (up to
//    2,000,000,000), and +5% attack per level over 1.
// 'state: 0 waiting, 1 fighting, 2 corrected.
1@ba_go,1,1,0	script	#wsg_hol	-1,{
	end;
OnInstanceInit:
	'state = 0;
	disablenpc instance_npcname("Other World#wsg_hol");
	disablenpc instance_npcname("Way Back#wsg_hol");
	disablenpc instance_npcname("Dimensional Portal#wsg_hol");
	end;
OnStart:
	if ('state != 0)
		end;
	'state = 1;
	.@m$ = instance_mapname("1@ba_go");
	'level = cap_value($@wsg_hol_lv[instance_id()], 1, 20);
	$@wsg_hol_lv[instance_id()] = 0;
	monster .@m$,54,52,"Dimensional Criminal",22174,1,instance_npcname("#wsg_hol") + "::OnBoss";
	'boss = $@mobid[0];
	// Capped at 2,000,000,000 (level 17 and up).
	.@hp = 400000000 + 100000000 * min('level - 1, 16);
	setunitdata 'boss, UMOB_MAXHP, .@hp;
	setunitdata 'boss, UMOB_HP, .@hp;
	getunitdata 'boss, .@d;
	'atkmin = .@d[UMOB_ATKMIN] * (100 + 5 * ('level - 1)) / 100;
	'atkmax = .@d[UMOB_ATKMAX] * (100 + 5 * ('level - 1)) / 100;
	'def = .@d[UMOB_DEF];
	'mdef = .@d[UMOB_MDEF];
	setunitdata 'boss, UMOB_ATKMIN, 'atkmin;
	setunitdata 'boss, UMOB_ATKMAX, 'atkmax;
	'traps = 0;
	'passes = 0;
	'groggy = 0;
	'sanct = 0;
	'illusions = 0;
	mapannounce .@m$, "Sealing Spirit Stone: Barrier level " + 'level + ". The dimensional criminal wakes. Keep close to the centre of the prison: its traps grow worse the further out you stand.", bc_map, "0x88CCFF";
	.@t = 0;
	while ('state == 1) {
		sleep 1000;
		if ('state != 1 || !unitexists('boss))
			end;
		.@t++;
		getunitdata 'boss, .@d;
		// Groggy: counts down, then the criminal recovers its defenses.
		if ('groggy > 0) {
			'groggy--;
			if ('groggy == 0) {
				setunitdata 'boss, UMOB_DEF, 'def;
				setunitdata 'boss, UMOB_MDEF, 'mdef;
				mapannounce .@m$, "The dimensional criminal shakes off the blow.", bc_map, "0xFF4444";
			}
		} else if (.@t % 5 == 0) {
			// Regeneration: 0.01% of max HP per barrier level every 5 seconds,
			// and 0.02% more for each pass through the other world's portal.
			.@heal = .@d[UMOB_MAXHP] / 10000 * ('level + 2 * 'passes);
			setunitdata 'boss, UMOB_HP, min(.@d[UMOB_MAXHP], .@d[UMOB_HP] + .@heal);
		}
		// Sanctuary.
		if ('sanct > 0) {
			'sanct--;
			if (distance(.@d[UMOB_X], .@d[UMOB_Y], 'sx, 'sy) > 4 || 'sanct == 0) {
				'sanct = 0;
				setunitdata 'boss, UMOB_DAMAGETAKEN, 100;
				mapannounce .@m$, "The sanctuary around the dimensional criminal breaks.", bc_map, "0x88CCFF";
			}
		} else if (.@t % 45 == 0 && 'groggy == 0) {
			'sanct = 15;
			'sx = .@d[UMOB_X];
			'sy = .@d[UMOB_Y];
			setunitdata 'boss, UMOB_DAMAGETAKEN, 0;
			mapannounce .@m$, "The dimensional criminal declares a sanctuary! Nothing can touch it there. Pull it out!", bc_map, "0xFF4444";
		}
		// Traps.
		if (.@t % 8 == 0 && 'groggy == 0)
			callsub S_Traps, .@m$;
	}
	end;

S_Traps:
	.@m$ = getarg(0);
	.@hit = 0;
	.@n = getmapunits(BL_PC, .@m$, .@pc);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (getmapxy(.@mm$, .@x, .@y, BL_PC, .@pc[.@i]) != 0 || .@x > 200)
			continue;
		.@far = distance(.@x, .@y, 54, 49);
		if (.@far < 8)
			continue;
		if (attachrid(.@pc[.@i])) {
			specialeffect2 EF_THUNDERSTORM2;
			if (Hp > 0)
				percentheal -min(25, 5 + .@far / 2), 0;
			detachrid;
			.@hit++;
		}
	}
	if (.@hit == 0)
		return;
	'traps += .@hit;
	// Every trap hit: +3% attack, up to +150%.
	.@pct = 100 + min(150, 3 * 'traps);
	setunitdata 'boss, UMOB_ATKMIN, 'atkmin * .@pct / 100;
	setunitdata 'boss, UMOB_ATKMAX, 'atkmax * .@pct / 100;
	if ('traps >= 10 && 'illusions == 0) {
		'illusions = 3;
		.@e$ = instance_npcname("#wsg_hol") + "::OnIllusion";
		.@ihp = 20000000 * 'level;
		setarray .@ix, 343, 361, 352;
		setarray .@iy, 49, 49, 60;
		for (.@i = 0; .@i < 3; .@i++) {
			monster .@m$,.@ix[.@i],.@iy[.@i],"Criminal's Illusion",22175,1,.@e$;
			setunitdata $@mobid[0], UMOB_MAXHP, .@ihp;
			setunitdata $@mobid[0], UMOB_HP, .@ihp;
		}
		enablenpc instance_npcname("Other World#wsg_hol");
		enablenpc instance_npcname("Way Back#wsg_hol");
		mapannounce .@m$, "The traps have torn the prison open: a portal to the other world glows at the north wall. Its illusions are bound to the criminal. Few and strong should go; every crossing makes it heal faster.", bc_map, "0x88CCFF";
	}
	return;

OnIllusion:
	'illusions--;
	if ('illusions > 0)
		end;
	.@m$ = instance_mapname("1@ba_go");
	disablenpc instance_npcname("Other World#wsg_hol");
	disablenpc instance_npcname("Way Back#wsg_hol");
	areawarp .@m$,333,30,371,68,.@m$,54,60;
	if ('state != 1 || !unitexists('boss))
		end;
	'groggy = 20;
	'traps = 0;
	'sanct = 0;
	setunitdata 'boss, UMOB_DAMAGETAKEN, 100;
	setunitdata 'boss, UMOB_ATKMIN, 'atkmin;
	setunitdata 'boss, UMOB_ATKMAX, 'atkmax;
	setunitdata 'boss, UMOB_DEF, 'def / 10;
	setunitdata 'boss, UMOB_MDEF, 'mdef / 10;
	mapannounce .@m$, "The illusions shatter, and the dimensional criminal reels! Its traps fall silent and its defenses crack. Strike now!", bc_map, "0x88CCFF";
	end;

OnBoss:
	'state = 2;
	.@m$ = instance_mapname("1@ba_go");
	killmonster .@m$, instance_npcname("#wsg_hol") + "::OnIllusion";
	disablenpc instance_npcname("Other World#wsg_hol");
	disablenpc instance_npcname("Way Back#wsg_hol");
	areawarp .@m$,333,30,371,68,.@m$,54,44;
	mapannounce .@m$, "The dimensional criminal is corrected. For a moment its face is not Rigel's at all. A dimensional portal opens: report there.", bc_map, "0x88CCFF";
	enablenpc instance_npcname("Dimensional Portal#wsg_hol");
	end;
}

			close;
		}
		callfunc "F_WSG_SetBarrier", .@lv + 1;
		wsg_hol = 0;
		mes "[Dimensional Portal]";
		mes "The barrier on your protection is now level " + (.@lv + 1) + ".";
		close2;
		break;
	case 3:
		close2;
		break;
	}
	warp "t_garden",165,228;
	end;

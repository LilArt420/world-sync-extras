//===== World Sync Extras: Garden of Time block ==============
//= The Subspace vending machines south of Rigel's Sealing Spirit Stone,
//= rebuilt in dialogue. rAthena's #2 opens kRO's Item Reform window for
//= the Divine Beast's Cloak upgrade, which our client can't show (the
//= player is left stuck: "finish the task in progress"); #3 and #4 open
//= enchant tables rAthena ships commented out, so nothing happens; and
//= #4 is never uncloaked at all. These copies stand on the same spots,
//= sell the same barters and do the upgrade and the enchants with the
//= official tables (npc/wse_dialog.txt). #1 (the exchanges) works and
//= stays rAthena's.
//=
//= The four Functional Vending Machines (tools, storage, save point) are
//= copied too: rAthena's "Tool Dealer" opens the SELL window (callshop
//= type 2) of a shop holding only Red Potions, so players saw their own
//= inventory and took it for the machine's stock. Ours sells the usual
//= tools.
//============================================================

t_garden,1,1,0	script	#wsg_vending	-1,{
	end;
OnInit:
	setarray .@n$, "Subspace vending machine#2", "Subspace vending machine#3", "Subspace vending machine#4",
		"Functional Vending Machine#1", "Functional Vending Machine#2", "Functional Vending Machine#3", "Functional Vending Machine#4";
	for (.@i = 0; .@i < getarraysize(.@n$); .@i++) {
		if (getnpcid(0, .@n$[.@i]))
			disablenpc .@n$[.@i];
	}
	end;
}

// Equipment: the accessories, the cloaks, and the cloak upgrade.
t_garden,132,120,2	script(CLOAKED)	Subspace Vending Machine#wsg2	HEROS_IN_ORB_2,{
	if (isbegin_quest(12620) != 2)
		end;
	if (checkweight(1201,1) == 0) {
		mes "I cannot communicate with the barrier because I have too many types or weights of objects.";
		close;
	}
	mes "[Subspace Vending Machine]";
	mes "This unit specializes in equipment. What kind of deal do you want?";
	next;
	switch (select("Purchase Circulation of Life accessories.:Purchase a Divine Beast's Cloak.:Upgrade a Divine Beast's Cloak.:Cancel.")) {
	case 1:
		close2;
		callshop "barter_t_garden_4";
		end;
	case 2:
		close2;
		callshop "barter_t_garden_5";
		end;
	case 3:
		callfunc "F_WSE_Reform", 102514, "[Subspace Vending Machine]";	// Season_H_Upgrade
		end;
	}
	close;
OnInit:
	setunittitle getnpcid(0), "<Clothing Upgrade Specialist>";
	end;
}

t_garden,156,222,2	script	Functional Vending Machine#wsgf1	4_VENDING_MACHINE2,{
	// A dialog box first: close2 needs its Close button. With a bare menu
	// our client has no box to put it in, and the player is stuck until
	// they relog ("finish the task in progress").
	mes "[Functional Vending Machine]";
	mes "Welcome. Please choose a service.";
	next;
	switch (select("Tool Dealer.:Storage (100z).:Use the save point service.:Cancel.")) {
	case 1:
		close2;
		callshop "wsg_tools", 1;
		end;
	case 2:
		if (Zeny < 100) {
			mes "[Functional Vending Machine]";
			mes "To use the Storage, please insert 100 zeny.";
			close;
		}
		Zeny -= 100;
		mes "[Functional Vending Machine]";
		mes "I will open your storage. Thank you for using it.";
		close2;
		openstorage;
		end;
	case 3:
		// Each machine saves next to itself, as rAthena's do.
		.@n = atoi(charat(strnpcinfo(2), getstrlen(strnpcinfo(2)) - 1));
		setarray .@x, 0, 156, 97, 90, 137;
		setarray .@y, 0, 220, 135, 115, 106;
		savepoint "t_garden", .@x[.@n], .@y[.@n];
		mes "[Functional Vending Machine]";
		mes strcharinfo(0) + ", your location has been safely saved.";
		close;
	}
	close;
OnInit:
	questinfo QTYPE_CLICKME, QMARK_YELLOW, "isbegin_quest(12592) == 2";
	end;
}

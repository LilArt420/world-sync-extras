// The Pet Mimics of the Unknown Basement: one of them, picked when
// the errand starts, is the Devil's pet; the others point the way.
-	script	Pet Mimic#wsi	-1,{
	.@me = atoi(substr(strnpcinfo(2), 3, 4));
	if (isbegin_quest(5969) != 1 || wsi_mimic < 1) {
		mes "- A Mimic is sniffing around.";
		mes "It eyes your bag hungrily and";
		mes "snaps its lid. Better keep";
		mes "your distance. -";
		close;
	}
	if (.@me != wsi_mimic) {
		setarray .@x[1], 77, 267, 244, 136, 110, 195, 111, 37, 20, 37;
		setarray .@y[1], 79, 180, 198, 143, 173, 269, 259, 221, 158, 70;
		.@dx = .@x[wsi_mimic] - .@x[.@me];
		.@dy = .@y[wsi_mimic] - .@y[.@me];
		if (.@dy > 20)
			.@dir$ = "north";
		else if (.@dy < -20)
			.@dir$ = "south";
		if (.@dx > 20)
			.@dir$ = .@dir$ + "east";
		else if (.@dx < -20)
			.@dir$ = .@dir$ + "west";
		if (.@dir$ == "")
			.@dir$ = "close by";
		mes "- This Mimic has no collar:";
		mes "a wild one, not the Devil's";
		mes "pet. It snaps at the lunch box";
		mes "and misses. -";
		next;
		mes "- Small lid-shaped prints in";
		mes "the dust lead ^0000FF" + .@dir$ + "^000000. -";
		close;
	}
	mes "- A Mimic with a little ^FF0000red";
	mes "collar^000000! The moment you open the";
	mes "lunch box, it lunges, gobbles";
	mes "every crumb and nips your";
	mes "fingers for good measure. -";
	next;
	mes "- ...Well, it ate. That counts.";
	mes "Time to report to the Devil";
	mes "in the Box. -";
	erasequest 5969;
	setquest 5970;
	close;
}

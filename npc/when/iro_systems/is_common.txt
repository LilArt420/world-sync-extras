// A daily or weekly whose cooldown quest is still running: 1, or 0
// when it's over (or never started; the expired quest is dropped).
// callfunc("F_WSI_Waiting", <cooldown quest>)
function	script	F_WSI_Waiting	{
	.@q = getarg(0);
	if (isbegin_quest(.@q) == 0)
		return 0;
	if (checkquest(.@q, PLAYTIME) == 2) {
		erasequest .@q;
		return 0;
	}
	return 1;
}

// Starts a cooldown quest afresh.
// callfunc "F_WSI_Cooldown", <cooldown quest>;
function	script	F_WSI_Cooldown	{
	.@q = getarg(0);
	if (isbegin_quest(.@q))
		erasequest .@q;
	setquest .@q;
	return;
}

// The World Sync doors this block's NPCs stand in for: the
// "entrances" setting's portal into the Clock Tower Unknown Basement
// (the Devil in the Box sends you down and checks the level).
-	script	WSI_Loader	-1,{
	end;
OnInit:
	setarray .@n$, "ws_c_tower1_26250";
	for (.@i = 0; .@i < getarraysize(.@n$); .@i++) {
		if (getnpcid(0, .@n$[.@i]))
			disablenpc .@n$[.@i];
	}
	end;
}

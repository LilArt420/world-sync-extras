prt_church,186,36,4	script	Acolyte Leader Alice#wsi	4_F_ACOLYTE,{
	.@acolyte = ((eaclass() & (EAJ_BASEMASK | EAJL_2)) == EAJ_ACOLYTE);
	if (!.@acolyte) {
		mes "[Alice Yisha]";
		mes "Welcome to the Cathedral.";
		mes "I look after the young";
		mes "Acolytes of Prontera.";
		cutin "acact_01.bmp", 2;
		next;
		mes "[Alice Yisha]";
		mes "We train them in the old";
		mes "ship that sank off Izlude.";
		mes "Damp, dark and full of";
		mes "restless bones: the perfect";
		mes "classroom! May Odin bless you.";
		close2;
		cutin "", 255;
		end;
	}
	if (isbegin_quest(11345) < 2) {
		mes "[Alice Yisha]";
		mes "Oh, Odin! An Acolyte!";
		mes "Do you remember me? Alice";
		mes "Yisha, from the Criatura";
		mes "Academy. I'm in charge of";
		mes "the Acolytes here now.";
		cutin "acact_01.bmp", 2;
		if (isbegin_quest(11345) == 0)
			setquest 11345;
		completequest 11345;
		next;
		mes "[Alice Yisha]";
		mes "Prayer is important, but an";
		mes "Acolyte must also be able to";
		mes "stand on their own two feet.";
		mes "So we train in the ^0000FFIzlude";
		mes "Sunken Ship^000000.";
		cutin "acact_02.bmp", 2;
		next;
		mes "[Alice Yisha]";
		mes "A lamb named ^0000FFLaforet^000000 keeps";
		mes "the entrance. She'll bring";
		mes "you back here when you're";
		mes "done. Talk to me whenever";
		mes "you're ready to train.";
		close2;
		cutin "", 255;
		end;
	}
	setarray .@q[0], 11346, 11347, 11348;
	setarray .@lv[0], 13, 18, 23;
	setarray .@task$[0],
		"defeat ^0000FF20 Roda Frogs^000000 and ^0000FF20 Dead Plankton^000000 on the wreck's 1st floor",
		"defeat ^0000FF30 Weak Skeletons^000000, the soft-boned ones",
		"defeat ^0000FF30 Weak Soldier Skeletons^000000, the hard-boned ones";
	for (.@i = 0; .@i < 3; .@i++)
		if (isbegin_quest(.@q[.@i]) < 2)
			break;
	mes "[Alice Yisha]";
	mes "Welcome back, " + strcharinfo(0) + ".";
	mes "How may I help you?";
	cutin "acact_01.bmp", 2;
	next;
	switch (select("Training in the shipwreck:Take me to the shipwreck:Nothing, thank you")) {
	case 1:
		if (.@i == 3) {
			mes "[Alice Yisha]";
			mes "You've finished all of my";
			mes "training! I'm so proud.";
			mes "Odin will surely be pleased";
			mes "with you, too.";
			cutin "acact_02.bmp", 2;
			close2;
			cutin "", 255;
			end;
		}
		if (isbegin_quest(.@q[.@i]) == 1) {
			if (checkquest(.@q[.@i], HUNTING) != 2) {
				mes "[Alice Yisha]";
				mes "Your training: " + .@task$[.@i] + ".";
				mes "Come back to me when it's done.";
				cutin "acact_03.bmp", 2;
				close2;
				cutin "", 255;
				end;
			}
			mes "[Alice Yisha]";
			mes "You did it! Your faith is";
			mes "strong, and so is your arm.";
			mes "Take these with my blessing.";
			cutin "acact_02.bmp", 2;
			completequest .@q[.@i];
			setarray .@base[0], 2000, 5000, 10000;
			setarray .@job[0], 1500, 3000, 6000;
			getexp .@base[.@i], .@job[.@i];
			getitem 523, 5; // Holy Water
			close2;
			cutin "", 255;
			end;
		}
		if (BaseLevel < .@lv[.@i]) {
			mes "[Alice Yisha]";
			mes "The next lesson is for";
			mes "Acolytes of ^0000FFbase level " + .@lv[.@i] + "^000000.";
			mes "Train a little more first.";
			cutin "acact_03.bmp", 2;
			close2;
			cutin "", 255;
			end;
		}
		mes "[Alice Yisha]";
		mes "Here is your next lesson:";
		mes .@task$[.@i] + ".";
		cutin "acact_02.bmp", 2;
		next;
		if (select("I'll do it:Not now") == 2) {
			mes "[Alice Yisha]";
			mes "Whenever you're ready.";
			close2;
			cutin "", 255;
			end;
		}
		setquest .@q[.@i];
		mes "[Alice Yisha]";
		mes "May Odin guide your mace!";
		mes "Shall I send you to the";
		mes "shipwreck right away?";
		next;
		if (select("Yes, please:I'll walk") == 2) {
			mes "[Alice Yisha]";
			mes "Then talk to me when you";
			mes "want to go.";
			close2;
			cutin "", 255;
			end;
		}
		break;
	case 2:
		mes "[Alice Yisha]";
		mes "Watch your step: the deck";
		mes "is slippery. Off you go!";
		break;
	case 3:
		mes "[Alice Yisha]";
		mes "May Odin be with you.";
		close2;
		cutin "", 255;
		end;
	}
	close2;
	cutin "", 255;
	warp "treasure_n1", 163, 43;
	end;
}

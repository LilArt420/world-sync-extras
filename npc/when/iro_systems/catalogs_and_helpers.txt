//= Vending search in the markets, Prontera's Catalogue Magician,
//= Izlude's Increase AGI acolytes and the east library's Monster
//= Encyclopedias.
//===== Description: =========================================
//= Catalog Dispensers: like iRO's Shopping Boards (iROWiki,
//= Commerce), one search of the stalls on this map per click,
//= with the shop opened from the result list.
//= Catalogue Magician: iRO's spot (prontera 99,64) for rAthena's
//= official Catalog Magician (Window Shopper Catalogue, 200z).
//= Monster Encyclopedias: the same books as rAthena's in the old
//= library (prt_in).
//= Estimate: the acolytes' Increase AGI (level 10, 240 seconds).
//============================================================

-	script	Catalog Dispenser#wsi	-1,{
	mes "[Catalog Dispenser]";
	mes "Welcome to the market!";
	mes "This dispenser keeps the";
	mes "market's catalog up to date:";
	mes "every stall here, what it";
	mes "sells and what it buys.";
	next;
	if (select("Search the market's stalls:Cancel") == 2) {
		mes "[Catalog Dispenser]";
		mes "Happy shopping!";
		close;
	}
	close2;
	searchstores 1, SEARCHSTORE_EFFECT_REMOTE;
	end;
}

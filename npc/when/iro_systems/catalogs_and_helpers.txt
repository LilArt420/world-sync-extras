//= Catalog Dispensers: like iRO's Shopping Boards (iROWiki,
//= Commerce), one search of the stalls on this map per click,
//= with the shop opened from the result list.
//= The Catalogue Magician and the east library's Monster
//= Encyclopedias are rAthena's own scripts: see iro_ports.txt.
//= Estimate: the acolytes' Increase AGI (level 10, 240 seconds).

-	script	Catalog Dispenser#wsi	-1,{
	mes "[Catalog Dispenser]";
	mes "Welcome to the market!";
	mes "This dispenser keeps the";
	mes "market's catalog up to date:";
	mes "every stall here, what it";
	mes "sells and what it buys.";
	next;
	if (select("Search the market's stalls:Cancel") == 2) {
		mes "[Catalog Dispenser]";
		mes "Happy shopping!";
		close;
	}
	close2;
	searchstores 1, SEARCHSTORE_EFFECT_REMOTE;
	end;
}

-	script	Increase Agi Aco#wsi	-1,{
	mes "[Acolyte]";
	mes "May the blessings of Odin";
	mes "speed your steps, adventurer.";
	mes "Shall I cast Increase AGI on you?";
	next;
	if (select("Yes, please:No, thank you") == 2) {
		mes "[Acolyte]";
		mes "Then go in peace, and come";
		mes "back whenever you need me.";
		close;
	}
	mes "[Acolyte]";
	mes "Odin, grant this traveler";
	mes "swift feet!";
	specialeffect2 EF_INCAGILITY;
	sc_start SC_INCREASEAGI, 240000, 10;
	close;
}

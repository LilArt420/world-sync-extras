function	script	F_WSV_C1Asked	{
	if (wsv_ch1 == 435 && !(wsv_ch1d & getarg(0))) {
		wsv_ch1d = wsv_ch1d | getarg(0);
		callfunc "F_WSV_Step", "wsv_ch1", 435, getarg(1), "";
		if ((wsv_ch1d & 7) == 7) {
			callfunc "F_WSV_Step", "wsv_ch1", 437, "", "23275";
			dispbottom "Nobody really knew Neumann. Report back to Maxim at the town hall.";
		}
	}
	return;
}

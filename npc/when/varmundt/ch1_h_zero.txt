// Sells a one-hour service: a timed quest the player can see, and its
// end in a character variable. Returns 1 when bought.
// callfunc("F_WSV_C1Timed", "<speaker>", <quest>, "<end variable>",
//          <base level>, <Mana Rings>, "<what>")
function	script	F_WSV_C1Timed	{
	.@n$ = getarg(0);
	.@q = getarg(1);
	.@v$ = getarg(2);
	.@cost = getarg(4);
	if (BaseLevel < getarg(3)) {
		mes .@n$;
		mes "Not yet. The " + getarg(5) + " would tear apart before it settled on you. Come back at base level " + getarg(3) + ".";
		return 0;
	}
	.@left = getd(.@v$) - gettimetick(2);
	if (.@left > 0 && isbegin_quest(.@q) == 1) {
		mes .@n$;
		mes "Your " + getarg(5) + " is still holding: about " + ((.@left + 59) / 60) + " minute(s) left.";
		mes "I can start a new hour now for another " + .@cost + " " + getitemname(1001996) + ".";
		next;
		if (select("Start a new hour.:I'll keep this one.") == 2)
			return 0;
	}
	if (countitem(1001996) < .@cost) {
		mes .@n$;
		mes "That takes " + .@cost + " " + getitemname(1001996) + ". You have " + countitem(1001996) + ".";
		return 0;
	}
	delitem 1001996, .@cost;
	if (isbegin_quest(.@q))
		erasequest .@q;
	setquest .@q;
	setd .@v$, gettimetick(2) + 3600;
	specialeffect2 EF_BLESSING;
	return 1;
}

// Robin's dailies: Clean Up Zero Cell
// ------------------------------------------------------------
function	script	F_WSV_ZCDailies	{
	setarray .@q, 64261, 64263, 64265, 64267;
	setarray .@sec$, "Lux", "Nox", "Vita", "Ark";
	setarray .@cnt, 300, 300, 100, 100;
	// Turn-ins first.
	for (.@i = 0; .@i < 4; .@i++) {
		if (isbegin_quest(.@q[.@i]) != 1 || checkquest(.@q[.@i], HUNTING) != 2)
			continue;
		mes "[Robin]";
		mes "Zero Cell " + .@sec$[.@i] + " is quieter already. That buys us a day, at least.";
		completequest .@q[.@i];
		erasequest .@q[.@i];
		if (isbegin_quest(.@q[.@i] + 1))
			erasequest .@q[.@i] + 1;
		setquest .@q[.@i] + 1;
		questinfo_refresh;
		if (.@i < 2) {
			getitem 105161, 1;	// Tangled Singularity
			getitem 1001996, 15;	// Ch1_Mana_Ring
			callfunc "F_WSV_C2Reward", 2, 0;	// 2% of the level
		} else {
			getitem 105162, 1;	// Boundary (Frontier) Singularity
			getitem 1001996, 20;	// Ch1_Mana_Ring
			callfunc "F_WSV_C2Reward", 3, 0;	// 3% of the level
		}
		close;
	}
	mes "[Robin]";
	mes "The monsters in Zero Cell keep multiplying. If we don't thin them out every day, they'll spill into the zone. Get a seal from Newt first.";
	next;
	for (.@i = 0; .@i < 4; .@i++) {
		.@what$ = "Clean Up Zero Cell " + .@sec$[.@i] + " (" + .@cnt[.@i] + " monsters)";
		if (isbegin_quest(.@q[.@i]) == 1)
			.@menu$ += "^aaaaaa" + .@what$ + " (in progress)^000000:";
		else if (callfunc("F_WSV_Waiting", .@q[.@i] + 1))
			.@menu$ += "^aaaaaa" + .@what$ + " (done today)^000000:";
		else
			.@menu$ += .@what$ + ":";
	}
	.@i = select(.@menu$ + "Nothing today.") - 1;
	if (.@i >= 4)
		close;
	if (isbegin_quest(.@q[.@i]) == 1) {
		mes "[Robin]";
		mes "Still on that one: " + .@cnt[.@i] + " monsters in Zero Cell " + .@sec$[.@i] + ".";
		close;
	}
	if (callfunc("F_WSV_Waiting", .@q[.@i] + 1)) {
		mes "[Robin]";
		mes "That one's done for today. Come back after 4 AM.";
		close;
	}
	if (isbegin_quest(.@q[.@i]) == 2)
		erasequest .@q[.@i];
	setquest .@q[.@i];
	questinfo_refresh;
	mes "[Robin]";
	mes "Thanks. Defeat " + .@cnt[.@i] + " monsters in Zero Cell " + .@sec$[.@i] + " and come back to me.";
	close;
}

		end;
	}
	addtimer .@left * 1000 + 500, strnpcinfo(3) + "::OnExpire";
	end;
OnExpire:
	.@m$ = strcharinfo(3);
	if (.@m$ != "ch1zero1" && .@m$ != "ch1zero2" && .@m$ != "ch1zero3" && .@m$ != "ch1zero4")
		end;
	.@left = callfunc("F_WSV_ZCLeft", .@m$);
	if (.@left > 0) {
		addtimer .@left * 1000 + 500, strnpcinfo(3) + "::OnExpire";
		end;
	}
	dispbottom "Your mana resistance seal has worn off. The mana of Zero Cell pushes you out.";
	warp "ch1fild1", 104, 79;
	end;
}

// Seconds left on the seal that covers a Zero Cell map (0 if none).
// callfunc("F_WSV_ZCLeft", "<map>")
function	script	F_WSV_ZCLeft	{
	if (getarg(0) == "ch1zero1" || getarg(0) == "ch1zero2")
		.@t = wsv_zcln;
	else
		.@t = wsv_zcva;
	.@left = .@t - gettimetick(2);
	if (.@left < 0)
		.@left = 0;
	return .@left;
}

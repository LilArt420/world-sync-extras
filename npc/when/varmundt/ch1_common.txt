			"Clover#wsv_c1e", "620", "629";

// The end of a part of the chapter: experience for level 200+.
// Chapter 1's official rewards aren't published; these follow
// Episode 20's chapters in this mod (getexp2 60M).
// callfunc "F_WSV_C1Reward", <base exp>, <job exp>;
function	script	F_WSV_C1Reward	{
	getexp2 getarg(0), getarg(1);
	return;
}

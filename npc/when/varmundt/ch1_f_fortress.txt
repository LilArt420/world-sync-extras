// The fortress is a chain of floating platforms with no walkway between
// them: the outer spiral (where you come in, at its heart), the inner
// spiral, and the throne hall, a round platform below. Stairways of light
// appear when a platform is cleared.
1@ch1a,1,1,0	script	#wsv_md	-1,{
	end;
OnInstanceInit:
	'left = 16;
	disablenpc instance_npcname("Est#wsv_md");
	disablenpc instance_npcname("Upper Stairway#wsv_md");
	disablenpc instance_npcname("Throne Stairway#wsv_md");
	.@m$ = instance_mapname("1@ch1a");
	.@e$ = instance_npcname("#wsv_md") + "::OnLegion";
	// Up the outer spiral, from its heart to the top.
	setarray .@x, 282, 308, 256, 224;
	setarray .@y, 330, 283, 259, 306;
	for (.@i = 0; .@i < 4; .@i++) {
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Meat Cleaver Soldier",22460,2,.@e$;
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Hate Scythe Soldier",22461,1,.@e$;
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Crushing Plow Soldier",22462,1,.@e$;
	}
	end;
OnLegion:
	'left--;
	if ('left > 0)
		end;
	.@m$ = instance_mapname("1@ch1a");
	mapannounce .@m$, "Est: That's the outer guard down. A stairway of light has opened at the top of the spiral. Up!", bc_map, "0x88CCFF";
	enablenpc instance_npcname("Upper Stairway#wsv_md");
	// Down the inner spiral, from its top to its heart, where the officers wait.
	'left = 15;
	.@e$ = instance_npcname("#wsv_md") + "::OnOfficer";
	setarray .@x, 128, 155, 104, 71;
	setarray .@y, 330, 284, 259, 306;
	for (.@i = 0; .@i < 4; .@i++) {
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Meat Cleaver Soldier",22460,1,.@e$;
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Hate Scythe Soldier",22461,1,.@e$;
		areamonster .@m$,.@x[.@i]-3,.@y[.@i]-3,.@x[.@i]+3,.@y[.@i]+3,"Crushing Plow Soldier",22462,1,.@e$;
	}
	monster .@m$,117,296,"Immortal Legion Officer",22474,1,.@e$;
	monster .@m$,108,296,"Immortal Legion Officer",22475,1,.@e$;
	monster .@m$,112,298,"Vessel of the Immortal",22499,1,.@e$;
	end;
OnOfficer:
	'left--;
	if ('left > 0)
		end;
	.@m$ = instance_mapname("1@ch1a");
	mapannounce .@m$, "A cold voice rolls through the fortress: \"Who dares walk into my house?\"", bc_map, "0xFF4444";
	sleep 2000;
	mapannounce .@m$, "Est: Himmelmez. The throne hall is below the inner spiral; the stairway at its heart goes down. Go!", bc_map, "0x88CCFF";
	enablenpc instance_npcname("Throne Stairway#wsv_md");
	monster .@m$,85,150,"Himmelmez",22497,1,instance_npcname("#wsv_md") + "::OnBoss";
	end;
OnBoss:
	.@m$ = instance_mapname("1@ch1a");
	mapannounce .@m$, "Himmelmez: Impossible... I am... immortal...", bc_map, "0xFF4444";
	sleep 3000;
	mapannounce .@m$, "The air above Himmelmez's body tears open like cloth. A hand of shadow reaches out, takes the body, and pulls it through.", bc_map, "0xCCCCCC";
	sleep 3000;
	mapannounce .@m$, "A second voice, older and far colder: \"Not yet. You are still useful.\" The tear closes.", bc_map, "0xCCCCCC";
	enablenpc instance_npcname("Est#wsv_md");
	end;
}

hem_dun01,205,248,4	script	Mage Professor#wsv	10589,{
	if (wsv_ch1 < 1000) {
		mes "[Mage Professor]";
		mes "Hmm? Not now, not now. I'm measuring the fortress's magic.";
		close;
	}
	if (isbegin_quest(12660) == 0) {
		mes "[Mage Professor]";
		mes "Ah! You're the one who brought down Himmelmez. I've been wanting to meet you.";
		next;
		mes "[Mage Professor]";
		mes "The fortress kept a record of itself, the way a seashell keeps the sound of the sea. With enough energy, I can recreate the Dark Whisper as it was that day.";
		next;
		mes "[Mage Professor]";
		mes "Why? Because whatever took Himmelmez's body left a mark. If we replay the battle enough times, we might see a little more of it each time.";
		mes "The energy only lets me do it once every three days, mind you.";
		setquest 12660;
		completequest 12660;
		next;
	}
	if (isbegin_quest(12662) == 1) {
		mes "[Mage Professor]";
		mes "The recreation is still running. Step in and finish what you started.";
		next;
		if (select("Go in.:Not now.") == 2)
			close;
		callfunc "F_WSV_EnterMd", "Recreated Dark Whisper", "[Mage Professor]", "...The recreation collapsed. Try again in a moment.";
		end;
	}
	if (callfunc("F_WSV_Waiting", 12661)) {
		mes "[Mage Professor]";
		mes "The energy needs time to gather. Three days between recreations; I can't change physics.";
		close;
	}
	mes "[Mage Professor]";
	mes "Shall I recreate the Dark Whisper? You'll face Himmelmez again, as she was.";
	next;
	if (select("Recreate it.:Not now.") == 2)
		close;
	if (isbegin_quest(12662) == 2)
		erasequest 12662;
	setquest 12662;
	questinfo_refresh;
	callfunc "F_WSV_EnterMd", "Recreated Dark Whisper", "[Mage Professor]", "...The recreation collapsed. Try again in a moment.";
	end;
}

		mes "The Dark Whisper brews the sickness and calls the dead home. Cut the head off, and the body stops.";
		mes "The head is the ^0000FFImmortal Lord^000000. A creature called ^0000FFHimmelmez^000000, sitting in the deepest hall of the fortress.";
		next;
		mes "[Est]";
		mes "I've found a way in that the Legion doesn't guard: through the fortress's shadow, right here. It won't stay open long.";

		callfunc "F_WSV_Step", "wsv_ch1", 605, "12663", "12666";
		close3;
	}
	if (wsv_ch1 == 605 || wsv_ch1 == 610) {
		mes "[Est]";
		mes "The way stays open for an hour. Inside, fight your way through the Legion; the officers guard the inner hall, and Himmelmez waits at the far end.";
		mes "Ready?";
		next;
		if (select("Open the way.:Not yet.") == 2)
			close3;
		if (wsv_ch1 == 605)
			callfunc "F_WSV_Step", "wsv_ch1", 610, "12666", "12664";
		if (instance_enter("Destroy the Sky Fortress") != IE_OK) {
			.@id = instance_create("Destroy the Sky Fortress", IM_CHAR);
			if (.@id < 0 || instance_enter("Destroy the Sky Fortress") != IE_OK) {
				mes "[Est]";
				mes "...The shadow won't open. Try again in a moment.";
				close3;
			}
		}
		cutin "",255;
		end;
	}
	mes "[Est]";
	if (wsv_ch1 < 600)
		mes "Don't mind me. I'm watching the fortress.";
	else
		mes "The fortress still hangs there, but it's empty now. Quiet.";
	close3;

OnInit:
	questinfo( QTYPE_QUEST2, QMARK_YELLOW, "wsv_ch1 >= 600 && wsv_ch1 <= 610" );
	end;

hem_dun01,205,248,4	script	Mage Professor#wsv	10589,{
	if (wsv_ch1 < 1000) {
		mes "[Mage Professor]";
		mes "Hmm? Not now, not now. I'm measuring the fortress's magic.";
		close;
	}
	if (isbegin_quest(12660) == 0) {
		mes "[Mage Professor]";
		mes "Ah! You're the one who brought down Himmelmez. I've been wanting to meet you.";
		next;
		mes "[Mage Professor]";
		mes "The fortress kept a record of itself, the way a seashell keeps the sound of the sea. With enough energy, I can recreate the Dark Whisper as it was that day.";
		next;
		mes "[Mage Professor]";
		mes "Why? Because whatever took Himmelmez's body left a mark. If we replay the battle enough times, we might see a little more of it each time.";
		mes "The energy only lets me do it once every three days, mind you.";
		setquest 12660;
		completequest 12660;
		next;
	}
	if (isbegin_quest(12662) == 1) {
		mes "[Mage Professor]";
		mes "The recreation is still running. Step in and finish what you started.";
		next;
		if (select("Go in.:Not now.") == 2)
			close;
		if (instance_enter("Recreated Dark Whisper") != IE_OK) {
			.@id = instance_create("Recreated Dark Whisper", IM_CHAR);
			if (.@id < 0 || instance_enter("Recreated Dark Whisper") != IE_OK) {
				mes "[Mage Professor]";
				mes "...The recreation collapsed. Try again in a moment.";
				close;
			}
		}
		end;
	}
	if (callfunc("F_WSV_Waiting", 12661)) {
		mes "[Mage Professor]";
		mes "The energy needs time to gather. Three days between recreations; I can't change physics.";
		close;
	}
	mes "[Mage Professor]";
	mes "Shall I recreate the Dark Whisper? You'll face Himmelmez again, as she was.";
	next;
	if (select("Recreate it.:Not now.") == 2)
		close;
	if (isbegin_quest(12662) == 2)
		erasequest 12662;
	setquest 12662;
	questinfo_refresh;
	if (instance_enter("Recreated Dark Whisper") != IE_OK) {
		.@id = instance_create("Recreated Dark Whisper", IM_CHAR);
		if (.@id < 0 || instance_enter("Recreated Dark Whisper") != IE_OK) {
			mes "[Mage Professor]";
			mes "...The recreation collapsed. Try again in a moment.";
			close;
		}
	}
	end;
}

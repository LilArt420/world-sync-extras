// An apple for a resident (bits 1 Sirina, 2 Violas, 4 Gregor, 8 Artea;
// 16 Painu first). Ends the dialogue.
// callfunc "F_WSV_C2Apple", <bit>, "[Name]", "<thanks>";
function	script	F_WSV_C2Apple	{
	if (!(wsv_ch2b & 16)) {
		mes getarg(1);
		mes "Painu is handing out the apples, isn't he? Go to him first.";
		close3;
	}
	if (countitem(512) < 1) {
		mes getarg(1);
		mes "^4d4dffYou're out of " + mesitemlink(512) + "s.^000000";
		close3;
	}
	delitem 512,1;	// Apple
	wsv_ch2b = wsv_ch2b | getarg(0);
	mes getarg(1);
	mes getarg(2);
	if ((wsv_ch2b & 31) == 31) {
		next;
		mes getarg(1);
		mes "^4d4dffEveryone has had an apple. Artea is ready to listen.^000000";
		callfunc "F_WSV_C2Go", 600;
	}
	close3;
}

// Enters Nyrholt: getarg(0) 1 = the story, 2 = the Phantom of Nyrholt.
// A run of the other kind, or yesterday's phantom, is closed first.
// callfunc "F_WSV_C2Nyr", <mode>, "<speaker>";
function	script	F_WSV_C2Nyr	{
	.@key = getarg(0) * 1000 + ((getarg(0) == 2) ? gettime(DT_YEARDAY) : 0);
	.@id = instance_id(IM_CHAR);
	if (.@id > 0 && wsv_nyrk != .@key)
		instance_destroy .@id;
	wsv_nyrk = .@key;
	callfunc "F_WSV_EnterMd", "Nyrholt", getarg(1), "...The way won't open. Try again in a moment.";
	$@wsv_nyr[instance_id(IM_CHAR)] = getarg(0);
	end;
}

// ------------------------------------------------------------
// Memorial dungeon: Nyrholt (1@ch2a), the heart of the root.
// The story (Artea) and the daily Phantom of Nyrholt (Artea's
// phantom) share the map: $@wsv_nyr[<instance>] is 1 or 2.
// Four gates of Mysterious Intruders along the branching path, a
// portal at its north end, then the round arena: the Snapdragon (or
// its phantom) takes a tenth of the damage until it stands in one of
// Artea's three magic circles, which stun it for 10 seconds.
// ------------------------------------------------------------
1@ch2a	mapflag	nosave	SavePoint
1@ch2a	mapflag	noteleport
1@ch2a	mapflag	nomemo
1@ch2a	mapflag	nowarpto
1@ch2a	mapflag	nobranch
1@ch2a	mapflag	monster_noteleport

1@ch2a,1,1,0	script	#wsv_nyr	-1,{
	end;
OnInstanceInit:
	'state = 0;
	disablenpc instance_npcname("Heart of Nyrholt#wsv_nyr");
	disablenpc instance_npcname("Artea#wsv_nyr");
	end;
OnStart:
	if ('state != 0)
		end;
	'state = 1;
	'mode = cap_value($@wsv_nyr[instance_id()], 1, 2);
	$@wsv_nyr[instance_id()] = 0;
	.@m$ = instance_mapname("1@ch2a");
	// The intruders wear the shapes of the Flamebull, Paruson, Ice Needle
	// and Folnir Hre.
	setarray .@x, 209, 182, 235, 200;
	setarray .@y, 54, 69, 96, 110;
	setarray .@mob, 22685, 22694, 22701, 22704;
	'left = 16;
	for (.@g = 0; .@g < 4; .@g++) {
		areamonster .@m$, .@x[.@g] - 3, .@y[.@g] - 3, .@x[.@g] + 3, .@y[.@g] + 3, "Mysterious Intruder", .@mob[.@g], 4, instance_npcname("#wsv_nyr") + "::OnIntruder";
		for (.@i = 0; .@i < 4; .@i++) {
			setunitdata $@mobid[.@i], UMOB_MAXHP, ('mode == 1) ? 3000000 : 6000000;
			setunitdata $@mobid[.@i], UMOB_HP, ('mode == 1) ? 3000000 : 6000000;
		}
	}
	mapannounce .@m$, "Artea: Four gates, four bands of intruders. Clear them all, and the way to the heart opens at the north end.", bc_map, "0x88CCFF";
	end;
OnIntruder:
	'left--;
	if ('left % 4 == 0 && 'left > 0)
		mapannounce instance_mapname("1@ch2a"), "A gate is clear. " + ('left / 4) + " to go.", bc_map, "0x88CCFF";
	if ('left > 0)
		end;
	mapannounce instance_mapname("1@ch2a"), "Artea: The last gate is open! The heart of Nyrholt, at the north end. The Snapdragon is waiting.", bc_map, "0x88CCFF";
	enablenpc instance_npcname("Heart of Nyrholt#wsv_nyr");
	end;
OnBoss:
	'state = 3;
	mapannounce instance_mapname("1@ch2a"), "The Snapdragon wilts, and the foreign mana it hoarded drifts loose like pollen.", bc_map, "0x88CCFF";
	enablenpc instance_npcname("Artea#wsv_nyr");
	end;
OnFight:
	if ('state != 1)
		end;
	'state = 2;
	.@m$ = instance_mapname("1@ch2a");
	if ('mode == 1)
		monster .@m$, 84, 140, "Snapdragon", 22706, 1, instance_npcname("#wsv_nyr") + "::OnBoss";
	else
		monster .@m$, 84, 140, "Snapdragon Phantom", 22711, 1, instance_npcname("#wsv_nyr") + "::OnBoss";
	'boss = $@mobid[0];
	setunitdata 'boss, UMOB_DAMAGETAKEN, 10;
	mapannounce .@m$, "Artea: It's soaking up the blows! Lure it into one of my magic circles!", bc_map, "0xFF4444";
	setarray .@cx, 74, 94, 84;
	setarray .@cy, 132, 132, 148;
	.@stun = 0;
	.@rest = 0;
	while ('state == 2 && unitexists('boss)) {
		sleep 1000;
		if ('state != 2 || !unitexists('boss))
			end;
		getunitdata 'boss, .@d;
		if (.@stun > 0) {
			.@stun--;
			if (.@stun == 0) {
				setunitdata 'boss, UMOB_DAMAGETAKEN, 10;
				mapannounce .@m$, "The Snapdragon shakes itself awake.", bc_map, "0xFF4444";
				.@rest = 5;
			}
			continue;
		}
		if (.@rest > 0) {
			.@rest--;
			continue;
		}
		for (.@i = 0; .@i < 3; .@i++) {
			if (distance(.@d[UMOB_X], .@d[UMOB_Y], .@cx[.@i], .@cy[.@i]) <= 2) {
				.@stun = 10;
				setunitdata 'boss, UMOB_DAMAGETAKEN, 100;
				mapannounce .@m$, "Artea's magic circle flares! The Snapdragon is stunned: hit it now!", bc_map, "0x88CCFF";
				break;
			}
		}
	}
	end;
}

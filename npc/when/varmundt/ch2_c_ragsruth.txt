// Story: piles 3, 8 and 12 hold traces (bits 2048, 4096, 8192).
// Daily "Even a Handful of Magic": each pile once a day, a Ghost
// Flower half of the time (wsv_c2p, wsv_c2pd, wsv_c2gf).
function	script	F_WSV_C2Pile	{
	.@n = getarg(0);
	mes "[Pile of Grass]";
	mes "A ragged pile of pale grass.";
	if (wsv_ch2 == 100) {
		setarray .@trace, 0, 0, 0, 2048, 0, 0, 0, 0, 4096, 0, 0, 0, 8192;
		.@b = (.@n < getarraysize(.@trace)) ? .@trace[.@n] : 0;
		next;
		if (select("Search the grass.:Leave it.") == 2)
			close;
		close2;
		progressbar "ffffff", 2;
		if (.@b && !(wsv_ch2b & .@b)) {
			wsv_ch2b = wsv_ch2b | .@b;
			dispbottom "Under the grass: torn threads of magic, still warm, like someone lay here waiting.";
			if ((wsv_ch2b & 30720) == 30720)
				dispbottom "That's everything. Go back to Artea.";
		} else
			dispbottom "Just grass, and a beetle that's annoyed with you.";
		end;
	}
	if (isbegin_quest(64240) == 1) {
		.@today = gettime(DT_YEAR) * 1000 + gettime(DT_YEARDAY);
		if (wsv_c2pd != .@today) {
			wsv_c2pd = .@today;
			wsv_c2p = 0;
		}
		if (wsv_c2p & (1 << .@n)) {
			mes "You've already searched this one today.";
			close;
		}
		next;
		if (select("Look for Ghost Flowers.:Leave it.") == 2)
			close;
		close2;
		progressbar "ffffff", 2;
		wsv_c2p = wsv_c2p | (1 << .@n);
		if (wsv_c2gf < 5 && rand(2)) {
			wsv_c2gf++;
			specialeffect2 EF_SPR_PLANT;
			dispbottom "A Ghost Flower, glowing pale in the shadows of the grass. (" + wsv_c2gf + "/5)";
		} else
			dispbottom "No Ghost Flower in this one.";
		end;
	}
	close;
}

// ------------------------------------------------------------
// Nirhult: callfunc "F_WSV_C2Nirhult", <0 story | 1 illusion>, "<map>", <x>, <y>;
// He fights you and your party; whoever of the party is on the map
// when he falls gets the credit.
// ------------------------------------------------------------
function	script	F_WSV_C2Nirhult	{
	.@cid = getcharid(0);
	.@old = getd("$@wsv_nir_" + .@cid);
	if (.@old && unitexists(.@old)) {
		dispbottom "Nirhult is already here!";
		end;
	}
	.@name$ = getarg(0) ? "Illusion of Nirhult" : "Nirhult";
	.@gid = monster(getarg(1), getarg(2), getarg(3), .@name$, 22705, 1, "WSV_Nirhult::OnKill");
	setunitdata .@gid, UMOB_LEVEL, 240;
	setunitdata .@gid, UMOB_MAXHP, getarg(0) ? 30000000 : 15000000;
	setunitdata .@gid, UMOB_HP, getarg(0) ? 30000000 : 15000000;
	setd "$@wsv_nir_" + .@cid, .@gid;
	setd "$@wsv_niro_" + .@gid, .@cid;
	setd "$@wsv_nira_" + .@gid, getcharid(3);
	setd "$@wsv_nirp_" + .@gid, getcharid(1);
	setd "$@wsv_nirm_" + .@gid, getarg(0);
	specialeffect2 EF_LIGHTSPHERE;
	dispbottom "A cold presence gathers into a shape: " + .@name$ + "!";
	end;
}

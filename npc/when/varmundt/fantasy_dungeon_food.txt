// Rewards: bookmark and completion card, the book's random bonus
// item, and EXP or Fantasy Growth Potions depending on level.
function	script	F_WSV_DFReward	{
	.@n = 0;
	if (BaseLevel >= 170) .@n = 5;
	else if (BaseLevel >= 150) .@n = 2;
	else if (BaseLevel >= 130) .@n = 1;
	mes "[Ink]";
	mes "Readers get to choose their reward. Experience now, or Fantasy Growth Potions for later?";
	next;
	.@pick = 1;
	if (.@n)
		.@pick = select("Experience.:" + .@n + " " + getitemname(101404) + (.@n > 1 ? "s" : "") + ".");
	getitem 1000924, 1;	// VR_Bookmark
	getitem 1000925, 1;	// VR_Reading_Card
	getitem 1000952, 1;	// VR_Clover_Ring
	if (.@pick == 2)
		getitem 101404, .@n;	// Fantastic_Exp_Potion
	else
		callfunc "F_WSV_GrowthExp", (.@n ? .@n : 1);
	if (!(wsv_dfb & 1)) {
		wsv_dfb = wsv_dfb | 1;
		if (isbegin_quest(16667) == 1)
			completequest 16667;
	}
	if (wsv_dfb & 8) {
		// Kathryne pays her bet.
		getitem 1000952, 1;
	}
	callfunc "F_WSV_Step", "wsv_df", 0, "16690", "";
	callfunc "F_WSV_Erase", "16669,16670,16671,16672,16673,16674,16675,16676,16677,16678,16679,16680,16681,16682,16683,16684,16685,16686,16687,16688,16689,16690";
	wsv_dfb = wsv_dfb & 1;
	wsv_dft = 0;
	setquest 16668;
	return;
}

// The value of <n> Fantasy Growth Potions, as experience now.
function	script	F_WSV_GrowthExp	{
	.@e = NextBaseExp / 10;
	if (BaseLevel >= 200)
		.@e = NextBaseExp / 50;
	if (.@e < 1)
		.@e = 1;
	getexp2 .@e * getarg(0), 0;
	return;
}

// Fantasy (Job) Growth Potion: a tenth of the current level, a
// fiftieth from level 200 on (db/when/varmundt/item_db.yml).
// callfunc "F_WSV_GrowthPotion", <0 base | 1 job>;
function	script	F_WSV_GrowthPotion	{
	if (getarg(0)) {
		.@e = NextJobExp / 10;
		if (BaseLevel >= 200)
			.@e = NextJobExp / 50;
		getexp2 0, max(.@e, 1);
	} else {
		.@e = NextBaseExp / 10;
		if (BaseLevel >= 200)
			.@e = NextBaseExp / 50;
		getexp2 max(.@e, 1), 0;
	}
	return;
}

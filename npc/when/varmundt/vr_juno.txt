//= (ro.gnjoy.com updates 308 and 312, 2026): combat-learning
//= Virtual Record books in the annex of the Juno Library, read

//=
//= Variables: wsv_vrpass (passes), wsv_vrday (day the last set
//= was bought), wsv_vrtime (reading time ends), wsv_orb (orbs),

//============================================================

// Opens a VR book: callfunc "F_WSV_VRBook", "<title>", <level>, "<map>", <x>, <y>;
function	script	F_WSV_VRBook	{
	mes "[" + getarg(0) + "]";
	mes "A Fantasy Series book of the combat-learning collection. Recommended for base level " + getarg(1) + " or more.";
	if (BaseLevel < getarg(1)) {
		mes "^4d4dffThe pages won't open for you yet (base level " + getarg(1) + ").^000000";
		close;
	}
	next;
	if (wsv_vrtime > gettimetick(2)) {
		mes "Your reading authentication is still valid for " + ((wsv_vrtime - gettimetick(2)) / 60) + " minute(s).";
		next;
		if (select("Open the book.:Not now.") == 2)
			close;
	} else {
		if (wsv_vrpass < 1) {
			mes "^4d4dffYou need a Juno Library Pass. The Library Pass Guide sells them.^000000";
			close;
		}
		mes "Submit a Library Pass for an hour of reading? You have " + wsv_vrpass + ".";
		next;
		if (select("Submit a pass.:Not now.") == 2)
			close;
		wsv_vrpass--;
		wsv_vrtime = gettimetick(2) + 3600;
	}
	close2;
	specialeffect2 EF_TELEPORTATION2;
	warp getarg(2), getarg(3), getarg(4);
	end;
}

// callfunc "F_WSV_VRJudge", "<name>", <made bit>, <result shift>,
//          <materials quest>, <judged quest>, "<storage map>", "<kind>";
function	script	F_WSV_VRJudge	{
	.@n$ = "[" + getarg(0) + "]";
	.@b = getarg(1);
	.@sh = getarg(2);
	.@q = getarg(3);
	.@w = getarg(4);
	if (wsv_vrc & .@b) {
		.@r = (wsv_vrc >> .@sh) & 15;
		mes .@n$;
		setarray .@verdict$, "", "...This doesn't work. At all. But you tried, and the rules say I must give you something.", "It works. It's not remarkable, but it works. Issgard could use this.", "Oh, this is good. Practical, solid, clever. Well done!", "......This is a masterpiece. I want to take it home. Is that allowed? It's allowed.";
		setarray .@orbs, 0, 1, 3, 6, 10;
		mes .@verdict$[.@r];
		wsv_vrc = wsv_vrc &~ (.@b | (15 << .@sh));
		if (isbegin_quest(.@w))
			erasequest .@w;
		setquest .@w;
		callfunc "F_WSV_Orb", .@orbs[.@r];
		getexp2 15000000 * .@r, 7000000 * .@r;
		close3;
	}
	if (isbegin_quest(.@q) == 1) {
		mes .@n$;
		if (checkquest(.@q, HUNTING) == 2)
			mes "You have your materials. Now make your " + getarg(6) + " at the simulator, and bring it to me.";
		else
			mes "Gather the materials first: defeat 30 monsters in the " + (getarg(5) == "vr_zrr03" ? "Frozen Treasure Storage" : "Frozen Food Storage") + ".";
		close3;
	}
	if (callfunc("F_WSV_Waiting", .@w)) {
		mes .@n$;
		mes "One entry a day. I'll judge your next " + getarg(6) + " tomorrow.";
		close3;
	}
	mes .@n$;
	if (.@b == 1) {
		mes "The criterion is only one: is it an invention Issgard needs right now?";
		mes "Free and original ideas, please.";
	} else {
		mes "I want Issgard's own taste, made with Issgard's ingredients!";
	}
	next;
	if (select("Enter the contest.:Not today.") == 2)
		close3;
	if (isbegin_quest(.@q) == 2)
		erasequest .@q;
	setquest .@q;
	questinfo_refresh;
	mes .@n$;
	mes "Then go and get your materials: 30 monsters in the " + (getarg(5) == "vr_zrr03" ? "Frozen Treasure Storage" : "Frozen Food Storage") + ".";
	close3;
}

// callfunc "F_WSV_VRSim", <materials quest>, <made bit>, <result shift>, "<cut-in prefix>";
// Three parts are picked; today's theme wants parts 3t, 3t+1, 3t+2.
function	script	F_WSV_VRSim	{
	.@q = getarg(0);
	.@b = getarg(1);
	.@sh = getarg(2);
	.@cut$ = getarg(3);
	if (.@cut$ == "yuvr01_box_")
		setarray .@part$, "warm stone", "a coil", "a strap", "glow moss", "a glass case", "oil", "runners", "a gyro", "a rope", "wool", "a wind rune", "a clasp";
	else
		setarray .@part$, "seaweed", "snail meat", "salt", "sweet cells", "flour", "berries", "dried meat", "nuts", "a fluorescent cell", "calamari", "sour cells", "ice straw";
	mes "[AI Simulator]";
	if (wsv_vrc & .@b) {
		mes "A finished work is waiting in the output tray. Take it to the judge, or throw it in the trash can to start over.";
		close;
	}
	if (isbegin_quest(.@q) != 1 || checkquest(.@q, HUNTING) != 2) {
		mes "INSERT MATERIALS. Materials must be obtained from VR monsters by the contestant.";
		mes "^4d4dffEnter the contest with a judge, then gather the materials first.^000000";
		close;
	}
	mes "Materials detected. Select three components.";
	next;
	for (.@i = 0; .@i < 12; .@i++)
		.@menu$ += .@part$[.@i] + ":";
	.@a = select(.@menu$) - 1;
	.@b2 = select(.@menu$) - 1;
	.@c = select(.@menu$) - 1;
	.@t = callfunc("F_WSV_VRTheme");
	.@hit = (.@a / 3 == .@t) + (.@b2 / 3 == .@t) + (.@c / 3 == .@t);
	if (.@a == .@b2 || .@a == .@c || .@b2 == .@c)
		.@hit = min(.@hit, 1);
	mes "[AI Simulator]";
	mes "Components: " + .@part$[.@a] + ", " + .@part$[.@b2] + ", " + .@part$[.@c] + ".";
	mes "Processing...";
	close2;
	progressbar "ffffff", 4;
	.@r = 1 + .@hit;
	if (.@hit == 3 && rand(100) < 50)
		.@r = 4;
	else if (.@hit == 3)
		.@r = 3;
	setarray .@pic$, "", "fail", "norm", "succ01", "succ02";
	if (.@cut$ == "yuvr01_cook_") {
		setarray .@pic$, "", "fail", "norm", "norm", "epic01";
		if (.@r == 4)
			.@pic$[4] = (rand(2) ? "epic01" : "epic02");
	}
	cutin .@cut$ + .@pic$[.@r] + ".png", 4;
	completequest .@q;
	erasequest .@q;
	wsv_vrc = (wsv_vrc &~ (15 << .@sh)) | .@b | (.@r << .@sh);
	setarray .@word$, "", "a failure", "an ordinary work", "a fine work", "a masterpiece";
	dispbottom "The simulator produced " + .@word$[.@r] + ". Show it to the judge.";
	sleep2 3000;
	cutin "", 255;
	end;
}

// callfunc "F_WSV_VRFloor", <floor>, "<map>", <x>, <y>, <duty quest>;
function	script	F_WSV_VRFloor	{
	.@f = getarg(0);
	.@q = getarg(4);
	mes "[Clerk]";
	mes "This is the desk for the castle's floor " + .@f + ".";
	if (isbegin_quest(.@q) == 1 && checkquest(.@q, HUNTING) == 2) {
		mes "Fifty monsters cleared? The Hero will be pleased. Here's your pay.";
		completequest .@q;
		erasequest .@q;
		if (isbegin_quest(.@q + 3))
			erasequest .@q + 3;
		setquest .@q + 3;
		callfunc "F_WSV_Orb", 2 + .@f;
		getexp2 20000000 * .@f, 10000000 * .@f;
		close;
	}
	next;
	switch (select("Go up to floor " + .@f + ".:Take today's duty (clear 50 monsters).:Never mind.")) {
	case 1:
		close2;
		warp getarg(1), getarg(2), getarg(3);
		end;
	case 2:
		if (isbegin_quest(.@q) == 1 || callfunc("F_WSV_Waiting", .@q + 3)) {
			mes "[Clerk]";
			mes "You already have today's duty for this floor.";
			close;
		}
		if (isbegin_quest(.@q) == 2)
			erasequest .@q;
		setquest .@q;
		questinfo_refresh;
		mes "[Clerk]";
		mes "Fifty monsters on floor " + .@f + ". Come back here when you're done.";
		close;
	}
	close;
}

// callfunc "F_WSV_VRDuty", <quest>, <cooldown>, "<what>";
function	script	F_WSV_VRDuty	{
	if (isbegin_quest(getarg(0)) == 1) {
		mes "You're still on it: " + getarg(2);
		return;
	}
	if (callfunc("F_WSV_Waiting", getarg(1))) {
		mes "You've done that duty already. It resets at 4 AM.";
		return;
	}
	if (isbegin_quest(getarg(0)) == 2)
		erasequest getarg(0);
	setquest getarg(0);
	questinfo_refresh;
	mes getarg(2);
	return;
}

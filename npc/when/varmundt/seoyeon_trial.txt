		}
		mes "[Seoyeon]";
		mes "My name is Seoyeon. I watch over the old battlefields under the castle.";

function	script	F_WSV_SYTraces	{
	if (wsv_sy != 2 || wsv_sys != getarg(0)) {
		mes "Old footprints in the dirt, half washed away.";
		mes "Whoever left them is long gone.";
		close;
	}
	mes "Footprints... but they start nowhere and end nowhere.";
	mes "The soil around them is cold to the touch, and a torn strip of black cloth is caught on a root.";
	next;
	mes "You feel eyes on your back.";
	callfunc "F_WSV_Step", "wsv_sy", 3, "3522", "3523";
	specialeffect2 EF_SUMMONSLAVE;
	close;
}

function	script	F_WSV_SYFight	{
	.@s = getarg(0);
	mes "[Yeongwi]";
	mes "......";
	mes "You followed my steps. Few can.";
	next;
	mes "[Yeongwi]";
	mes "This forest burned, and I stayed. My clan fell, and I stayed.";
	mes "If you want me gone, take it from me with your blade.";
	next;
	if (select("Draw your weapon.:Step back.") == 2)
		close;
	.@old = getd("$@wsv_sy_" + getcharid(0));
	if (.@old && unitexists(.@old)) {
		mes "Yeongwi is already waiting for you, blade drawn!";
		close;
	}
	close2;
	setarray .@x, 55, 64, 123, 150, 160;
	setarray .@y, 162, 95, 15, 140, 39;
	cloakonnpcself "Yeongwi#wsv" + .@s;
	.@gid = monster("ama_dun02", .@x[.@s-1], .@y[.@s-1], "Yeongwi", 1401, 1, "WSV_Yeongwi::OnKill");
	setunitdata .@gid, UMOB_MAXHP, 60000;
	setunitdata .@gid, UMOB_HP, 60000;
	setunitdata .@gid, UMOB_LEVEL, 99;
	setd "$@wsv_sy_" + getcharid(0), .@gid;
	setd "$@wsv_syo_" + .@gid, getcharid(0);
	end;
}

-	script	WSV_Yeongwi	-1,{
	end;
OnKill:
	// Only the one who called him out frees his soul.
	.@cid = getd("$@wsv_syo_" + killedgid);
	setd "$@wsv_syo_" + killedgid, 0;
	if (.@cid)
		setd "$@wsv_sy_" + .@cid, 0;
	if (.@cid == getcharid(0) && wsv_sy == 3) {
		callfunc "F_WSV_Step", "wsv_sy", 4, "3523", "";
		dispbottom "Yeongwi's shadow breaks apart like smoke. Seoyeon has come running.";
		specialeffect2 EF_HOLYHIT;
	}
	end;
}

function	script	F_WSV_SYEnd	{
	if (wsv_sy == 3) {
		mes "[Seoyeon]";
		mes "That's him. Yeongwi.";
		mes "Don't hold back. He wouldn't want you to.";
		close;
	}
	if (wsv_sy != 4) {
		mes "[Seoyeon]";
		mes "...";
		close;
	}
	mes "[Seoyeon]";
	mes "It's over. Did you feel it, at the end? He bowed to you.";
	next;
	mes "[Seoyeon]";
	mes "My clan's stories say Yeongwi swore to guard this forest until someone stronger came along.";
	mes "I suppose that's you now.";
	next;
	mes "[Seoyeon]";
	mes "Something of him stays with whoever sets him free. His strength, his patience.";
	mes "Take it, and don't waste it.";
	.@e = min(NextBaseExp / 2, 2000000);
	.@j = min(NextJobExp / 2, 1000000);
	getexp2 .@e, .@j;
	callfunc "F_WSV_Step", "wsv_sy", 5, "", "";
	close;
}

// Who sees which copy: the spot's Yeongwi while hunting him, Seoyeon
// from the fight to the end.
function	script	F_WSV_SYScenes	{
	for (.@s = 1; .@s <= 5; .@s++) {
		if (wsv_sys == .@s && wsv_sy == 3)
			cloakoffnpcself "Yeongwi#wsv" + .@s;
		else
			cloakonnpcself "Yeongwi#wsv" + .@s;
		if (wsv_sys == .@s && (wsv_sy == 3 || wsv_sy == 4))
			cloakoffnpcself "Seoyeon#wsv" + .@s;
		else
			cloakonnpcself "Seoyeon#wsv" + .@s;
	}
	return;
}

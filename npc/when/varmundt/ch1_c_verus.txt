function	script	F_WSV_C1Talisman	{
	if (isbegin_quest(atoi(getarg(1))) == 1) {
		if (countitem(1001973) < 1) {
			mes "[Verus Believer]";
			mes "The song... the song...";
			next;
			mes "^4d4dffYou have no " + getitemname(1001973) + " left. Jeon Cheosa can write you new ones tomorrow.^000000";
			close;
		}
		mes "[Verus Believer]";
		mes "The song... the song won't stop... glory to the... glory...";
		next;
		delitem 1001973, 1;
		mes "You hang a " + getitemname(1001973) + " around the believer's neck.";
		mes "Their shaking slows. Their eyes clear, a little.";
		next;
		mes "[Verus Believer]";
		mes "...Quiet. It's quiet. Thank you. Thank you.";
		wsv_ch1c = wsv_ch1c | getarg(0);
		callfunc "F_WSV_Step", "wsv_ch1", wsv_ch1, getarg(1), "";
		if ((wsv_ch1c & 31) == 31) {
			callfunc "F_WSV_Step", "wsv_ch1", wsv_ch1, "", "17931";
			dispbottom "All five talismans given. Report to Jeon Cheosa at the Geffen Tower.";
		}
		close;
	}
	mes "[Verus Believer]";
	if (wsv_ch1 >= 350 && (wsv_ch1c & getarg(0)))
		mes "It's quiet today. I can almost think.";
	else
		mes "Glory... to the Legion... the song, the song...";
	close;
}

	mes "Flame Gold Coins! Flat as a pancake, shaped like a flame. Good in Yongjirak, good anywhere, good any age.";
	mes "You have ^0000FF" + wsv_flame + "^000000 coin(s).";
	next;
	setarray .@it, 1000322, 25731, 25729, 6635, 12208, 12118, 12119;
	setarray .@co, 2, 15, 15, 40, 3, 1, 1;
	for (.@i = 0; .@i < getarraysize(.@it); .@i++)
		.@menu$ += getitemname(.@it[.@i]) + " (" + .@co[.@i] + " coin" + (.@co[.@i] > 1 ? "s" : "") + "):";
	.@s = select(.@menu$ + "Never mind.") - 1;
	if (.@s >= getarraysize(.@it))
		close3;
	mes "[Flame Gold Coin Exchanger]";
	mes "How many ^0000FF" + getitemname(.@it[.@s]) + "^000000?";
	next;
	input .@n, 0, 100;
	if (.@n < 1)
		close3;
	if (wsv_flame < .@n * .@co[.@s]) {
		mes "[Flame Gold Coin Exchanger]";
		mes "Not enough coins, friend.";
		close3;
	}
	if (!checkweight(.@it[.@s], .@n)) {
		mes "[Flame Gold Coin Exchanger]";
		mes "You'll never carry all that.";
		close3;
	}
	wsv_flame -= .@n * .@co[.@s];
	getitem .@it[.@s], .@n;
	mes "[Flame Gold Coin Exchanger]";
	mes "Pleasure doing business.";
	close3;
}

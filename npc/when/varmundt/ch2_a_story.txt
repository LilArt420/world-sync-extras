// The three knots in Beledor.
function	script	F_WSV_C2Knot	{
	.@b = getarg(0);
	if (wsv_ch2 != 10 || (wsv_ch2b & .@b)) {
		mes "The air here is calm now.";
		close;
	}
	mes "The air is thick and humming. Threads of green light twist around each other, pulling tighter whenever you look away.";
	next;
	if (select("Ease the threads apart.:Leave it.") == 2)
		close;
	close2;
	progressbar "00ff00", 5;
	specialeffect2 EF_GREEN99_6;
	wsv_ch2b = wsv_ch2b | .@b;
	if ((wsv_ch2b & 7) == 7) {
		dispbottom "The last knot comes loose. Somewhere nearby, someone takes a deep breath.";
		callfunc "F_WSV_Step", "wsv_ch2", 20, "", "";
	} else
		dispbottom "The threads slide apart and fade. Another knot is somewhere in the sanctuary.";
	end;
}

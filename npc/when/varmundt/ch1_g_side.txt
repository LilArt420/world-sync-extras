// Jubilus in Volund Valley, after the story
// ------------------------------------------------------------
function	script	F_WSV_C1Jubilus	{
	if (wsv_ch1 < 700) {
		mes "[Jubilus]";
		if (wsv_ch1 < 515)
			mes "A visitor from Ash Village? Keep your regulator switched on. Muspelheim isn't kind to the unprepared.";
		else
			mes "The valley has more to teach than heat. I'm staying a while.";
		close3;
	}
	if (!(wsv_c1s & 2)) {
		if (isbegin_quest(24107) + isbegin_quest(24108) + isbegin_quest(24109) == 0) {
			mes "[Jubilus]";
			mes "You again! Good. I have Hal piling up and no idea what to do with half of it.";
			mes "The locals farm it, so they must use it somehow. Ask Harvan what Hal is good for, besides curing plagues.";
			callfunc "F_WSV_Step", "wsv_ch1", wsv_ch1, "24107", "24108";
			close3;
		}
		if (isbegin_quest(24109) == 1) {
			mes "[Jubilus]";
			mes "Farming! Hal feeds the soil, and the soil feeds the people. They need more Hal than they can gather, so...";
			mes "So we help. I'll post requests every day; you take whichever suits you.";
			wsv_c1s = wsv_c1s | 2;
			callfunc "F_WSV_Step", "wsv_ch1", wsv_ch1, "24109", "";
			getexp2 20000000, 10000000;
			close3;
		}
		mes "[Jubilus]";
		mes "Harvan is at the entrance of the valley, by his field.";
		close3;
	}
	// The four dailies: <quest>, <cooldown>, <what>.
	setarray .@q, 24110, 24112, 24114, 24116;
	setarray .@what$, "Grandpa Hal: bring 10 Hal from the valley's monsters", "Can It Be Grilled?: hunt 20 Volcano Clams", "Someone Wanted This?: hunt 20 Elixir Grass", "Tricked!!!: hunt 20 Rings of Fire";
	for (.@i = 0; .@i < 4; .@i++) {
		if (isbegin_quest(.@q[.@i]) != 1)
			continue;
		if (.@i == 0) {
			if (countitem(1001977) < 10)
				continue;
			delitem 1001977, 10;
		} else if (checkquest(.@q[.@i], HUNTING) != 2)
			continue;
		mes "[Jubilus]";
		mes "Done already? Wonderful. Harvan will be dancing in his field.";
		completequest .@q[.@i];
		erasequest .@q[.@i];
		setquest .@q[.@i] + 1;
		getexp2 25000000, 12000000;
		getitem 1001972, 2;	// Ch1_Root_Coin
		close3;
	}
	mes "[Jubilus]";
	mes "Today's requests. Take any you like.";
	next;
	for (.@i = 0; .@i < 4; .@i++) {
		if (isbegin_quest(.@q[.@i]) == 1)
			.@menu$ += "^aaaaaa" + .@what$[.@i] + " (in progress)^000000:";
		else if (callfunc("F_WSV_Waiting", .@q[.@i] + 1))
			.@menu$ += "^aaaaaa" + .@what$[.@i] + " (done today)^000000:";
		else
			.@menu$ += .@what$[.@i] + ":";
	}
	.@i = select(.@menu$ + "Nothing today.") - 1;
	if (.@i >= 4 || isbegin_quest(.@q[.@i]) == 1 || callfunc("F_WSV_Waiting", .@q[.@i] + 1))
		close3;
	if (isbegin_quest(.@q[.@i]) == 2)
		erasequest .@q[.@i];
	setquest .@q[.@i];
	questinfo_refresh;
	mes "[Jubilus]";
	mes "Thank you. Mind the lava.";
	close3;
}

// callfunc "F_WSV_C1Sister", "<name>", <bit>, "<interview quest>",
//          <freeing quest>, <daily first>, <daily repeat>, <cooldown>;
function	script	F_WSV_C1Sister	{
	.@n$ = "[" + getarg(0) + "]";
	.@b = getarg(1);
	if (wsv_c1br == 5 && !(wsv_c1bb & .@b)) {
		mes .@n$;
		if (.@b == 1) {
			mes "God loved us. That is why He chained us here, where nothing can touch us.";
			mes "Love is a chain too. Did you not know?";
		} else if (.@b == 2) {
			mes "We cannot feel the gods any more. That is the proof that we belong to them.";
			mes "Only what is kept is cut off from the world.";
		} else {
			mes "One thousand and four. One thousand and five. One thousand and four.";
			mes "...I am counting the days. I lose my place. I start again. Time here does not walk in a line.";
		}
		wsv_c1bb = wsv_c1bb | .@b;
		callfunc "F_WSV_Step", "wsv_c1br", 5, getarg(2), "";
		if ((wsv_c1bb & 7) == 7) {
			callfunc "F_WSV_Step", "wsv_c1br", 6, "8959", "8963";
			dispbottom "You've heard all three sisters. Il is waiting at the square of Brimir.";
		}
		close;
	}
	if (wsv_c1br == 8 && isbegin_quest(getarg(3)) == 1) {
		mes .@n$;
		mes "A petal? For me? ...It smells of a spring I don't remember.";
		next;
		mes "You press the Broken Petal against the chain. The iron softens like a stem, and slides away.";
		next;
		mes .@n$;
		mes "......";
		mes "My color didn't come back. But my feet are light.";
		if (getarg(3) < 8967)
			callfunc "F_WSV_Step", "wsv_c1br", 8, "" + getarg(3), "" + (getarg(3) + 1);
		else
			callfunc "F_WSV_Step", "wsv_c1br", 9, "8967", "8968";
		close;
	}
	if (wsv_c1br < 10) {
		mes .@n$;
		mes "......";
		close;
	}
	// Dailies, once the story has gone back to Ash Village.
	.@d1 = getarg(4);
	.@d2 = getarg(5);
	.@w = getarg(6);
	.@q = (isbegin_quest(.@d1) == 2 ? .@d2 : .@d1);
	if (isbegin_quest(.@q) == 1) {
		if (checkquest(.@q, HUNTING) != 2 || (.@b == 1 && countitem(1001975) < 20)) {
			mes .@n$;
			if (.@b == 1)
				mes "Fifty Tulips, fifty Calendulas, and twenty " + getitemname(1001975) + " to mourn them.";
			else if (.@b == 2)
				mes "Twenty Nyaiz. They keep coming for my face.";
			else
				mes "Ten Shadow Guardians. They were us, once.";
			close;
		}
		if (.@b == 1)
			delitem 1001975, 20;
		mes .@n$;
		mes "Thank you. Brimir is a little quieter tonight.";
		completequest .@q;
		if (isbegin_quest(.@w))
			erasequest .@w;
		setquest .@w;
		getexp2 25000000, 12000000;
		getitem 1001972, 2;	// Ch1_Root_Coin
		close;
	}
	if (callfunc("F_WSV_Waiting", .@w)) {
		mes .@n$;
		mes "Enough for today. Come back after the next dawn. If dawn still comes here.";
		close;
	}
	mes .@n$;
	if (.@b == 1)
		mes "The flowers were people once. Destroy fifty Tulips and fifty Calendulas, and bring me twenty " + getitemname(1001975) + ", so we can give them a grave. A formal grave.";
	else if (.@b == 2)
		mes "The Nyaiz keep scratching my face. Cats are cats, even when the world ends. Hunt twenty of them?";
	else
		mes "The Shadow Guardians... they were us. Will you destroy ten of them, so they can rest?";
	next;
	if (select("I'll do it.:Not today.") == 2)
		close;
	if (isbegin_quest(.@q) == 2) {
		erasequest .@q;
		.@q = .@d2;
	}
	if (isbegin_quest(.@q) == 2)
		erasequest .@q;
	setquest .@q;
	questinfo_refresh;
	close;
}

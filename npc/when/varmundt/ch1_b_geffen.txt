// Visiting a patient: callfunc "F_WSV_C1Visit", <bit>, <quest>;
function	script	F_WSV_C1Visit	{
	.@b = getarg(0);
	if (wsv_ch1 != 215 || (wsv_ch1b & .@b))
		return 0;
	wsv_ch1b = wsv_ch1b | .@b;
	callfunc "F_WSV_Step", "wsv_ch1", 215, getarg(1), "";
	if ((wsv_ch1b & 31) == 31) {
		callfunc "F_WSV_Step", "wsv_ch1", 220, "", "16917";
		dispbottom "You've met all five patients. Einhar and Gregor have arrived from Ash Village and are waiting by the clinic's entrance hall.";
	}
	return 1;
}

// 1 when the specimen request <quest> is open at the player's
// research reputation (the first five at 100, all ten at 300).
function	script	F_WSV_AbyssOpen	{
	if (wsv_abyss_rep >= 300)
		return 1;
	return (wsv_abyss_rep >= 100 && getarg(0) <= 16789);
}

// Gives or turns in one request: callfunc "F_WSV_AbyssQuest", <quest>, <cooldown quest>;
function	script	F_WSV_AbyssQuest	{
	.@q = getarg(0);
	.@w = getarg(1);
	mes "[Research Quest Manager]";
	if (isbegin_quest(.@q) == 1) {
		if (checkquest(.@q, HUNTING) != 2) {
			mes "You're still on this one. Check your quest log for the count.";
			return;
		}
		callfunc "F_WSV_AbyssDone", .@q;
		mes "Good work! The data looks clean. Here's your share.";
		return;
	}
	if (callfunc("F_WSV_Waiting", .@w)) {
		mes "You've already done that request today. It resets at 4 AM.";
		return;
	}
	if (isbegin_quest(.@q) == 2)
		erasequest .@q;
	setquest .@q;
	questinfo_refresh;
	if (.@q == 16779)
		mes "Hunt ^0000FF450 monsters^000000 of any kind in the Deep Abyss and report back to me.";
	else
		mes "Twenty-five of them, please. The specimen cases are labeled.";
	return;
}

// Rewards for a finished request.
function	script	F_WSV_AbyssDone	{
	.@q = getarg(0);
	completequest .@q;
	erasequest .@q;
	if (isbegin_quest(.@q + 1))
		erasequest .@q + 1;
	setquest .@q + 1;
	if (.@q == 16779) {
		getitem 1001554, 5;	// Abyss_Rune_Ore
		getitem 1001555, 2;	// Abyss_Rune
		getexp2 120000000, 70000000;
		.@rep = 30;
	} else {
		getitem 1001554, 2;
		getitem 1001555, 1;
		getexp2 20000000, 12000000;
		.@rep = 10;
	}
	.@old = wsv_abyss_rep;
	wsv_abyss_rep = min(wsv_abyss_rep + .@rep, 5000);
	dispbottom "Research reputation +" + .@rep + " (" + wsv_abyss_rep + ").";
	if (.@old < 100 && wsv_abyss_rep >= 100)
		dispbottom "The Research Quest Manager now has specimen requests for you.";
	if (.@old < 300 && wsv_abyss_rep >= 300)
		dispbottom "All ten specimen requests are open to you now.";
	if (wsv_abyss_rep >= 2000 && isbegin_quest(16801) != 2) {
		setquest 16801;
		completequest 16801;
		dispbottom "You've studied the Biosphere Deep Abyss in depth: Abyss Researcher.";
	}
	questinfo_refresh;
	return;
}

ba_chess,22,16,4	script	Sample Refiner Specialist#wsv	10345,{
	mes "[Sample Refiner Specialist]";
	mes "Bring me the shards and fragments you find in the Biosphere Deep Abyss and I'll refine them into jewels and runes.";
	next;
	// result, zeny, then up to three (material, amount) pairs
	setarray .@r, 1001552, 1001553, 1001554, 1001555, 1001556;
	setarray .@z, 10000, 10000, 20000, 30000, 50000;
	setarray .@m1, 1001550, 1001551, 1001552, 1001554, 1001555;
	setarray .@a1, 10, 10, 1, 3, 1;
	setarray .@m2, 0, 0, 1001553, 1001552, 1001553;
	setarray .@a2, 0, 0, 1, 5, 5;
	setarray .@m3, 0, 0, 0, 0, 1001554;
	setarray .@a3, 0, 0, 0, 0, 2;
	for (.@i = 0; .@i < getarraysize(.@r); .@i++)
		.@menu$ += getitemname(.@r[.@i]) + ":";
	.@i = select(.@menu$ + "Never mind.") - 1;
	if (.@i >= getarraysize(.@r))
		close;
	mes "[Sample Refiner Specialist]";
	mes "One ^0000FF" + getitemname(.@r[.@i]) + "^000000 takes:";
	mes "- " + .@a1[.@i] + " " + getitemname(.@m1[.@i]);
	if (.@m2[.@i]) mes "- " + .@a2[.@i] + " " + getitemname(.@m2[.@i]);
	if (.@m3[.@i]) mes "- " + .@a3[.@i] + " " + getitemname(.@m3[.@i]);
	mes "- " + callfunc("F_InsertComma", .@z[.@i]) + " zeny";
	next;
	mes "[Sample Refiner Specialist]";
	mes "How many? (up to 100)";
	next;
	input .@n, 0, 100;
	if (.@n < 1)
		close;
	if (countitem(.@m1[.@i]) < .@a1[.@i] * .@n
	|| (.@m2[.@i] && countitem(.@m2[.@i]) < .@a2[.@i] * .@n)
	|| (.@m3[.@i] && countitem(.@m3[.@i]) < .@a3[.@i] * .@n)
	|| Zeny < .@z[.@i] * .@n) {
		mes "[Sample Refiner Specialist]";
		mes "You don't have enough materials or zeny for that many.";
		close;
	}
	if (!checkweight(.@r[.@i], .@n)) {
		mes "[Sample Refiner Specialist]";
		mes "You won't be able to carry them. Lighten your bag first.";
		close;
	}
	delitem .@m1[.@i], .@a1[.@i] * .@n;
	if (.@m2[.@i]) delitem .@m2[.@i], .@a2[.@i] * .@n;
	if (.@m3[.@i]) delitem .@m3[.@i], .@a3[.@i] * .@n;
	Zeny -= .@z[.@i] * .@n;
	progressbar "ffffff", 2;
	getitem .@r[.@i], .@n;
	mes "[Sample Refiner Specialist]";
	mes "Done. Clean cuts, every one of them.";
	close;
}

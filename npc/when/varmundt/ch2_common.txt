// Who sees which Chapter 2 scene NPC (CLOAKED), by wsv_ch2.
function	script	F_WSV_C2Scenes	{
	.@m$ = strcharinfo(3);
	.@s = wsv_ch2;
	if (BaseLevel < 225)
		.@s = -1;
	if (.@m$ == "ygg_edge") {
		setarray .@t$,
			"Sasha#wsv_c2e", "0", "29";
	} else if (.@m$ == "ygg_fruit") {
		setarray .@t$,
			"El#wsv_c2f", "40", "9999";
	} else if (.@m$ == "veledor") {
		setarray .@t$,
			"Knot of Magic#wsv_c2a", "10", "19",
			"Knot of Magic#wsv_c2b", "10", "19",
			"Knot of Magic#wsv_c2c", "10", "19",
			"El#wsv_c2v", "20", "29";
	} else if (.@m$ == "mu_fild01") {
		setarray .@t$,
			"Chez#wsv_c2", "40", "59";
	} else if (.@m$ == "deadroot") {
		setarray .@t$,
			"Gregor#wsv_c2d", "60", "79",
			"Devries#wsv_c2d", "60", "79",
			"Frozen Harvan#wsv_c2a", "70", "79",
			"Frozen Harvan#wsv_c2b", "70", "79",
			"Frozen Harvan#wsv_c2c", "70", "79";
	} else if (.@m$ == "uknw_ruin") {
		setarray .@t$,
			"Gregor#wsv_c2b", "80", "89";
	} else if (.@m$ == "ch2safe4") {
		setarray .@t$,
			"Devries#wsv_c2s", "90", "129",
			"Gregor#wsv_c2s", "90", "129",
			"Devries#wsv_c2r", "1000", "9999",
			"Gregor#wsv_c2r", "1000", "9999";
	} else
		return;
	for (.@i = 0; .@i < getarraysize(.@t$); .@i += 3) {
		if (.@s >= atoi(.@t$[.@i+1]) && .@s <= atoi(.@t$[.@i+2]))
			cloakoffnpcself .@t$[.@i];
		else
			cloakonnpcself .@t$[.@i];
	}
	// Harvans already thawed stay gone; knots already loosened too.
	if (.@m$ == "deadroot") {
		setarray .@f$, "Frozen Harvan#wsv_c2a", "Frozen Harvan#wsv_c2b", "Frozen Harvan#wsv_c2c";
		for (.@i = 0; .@i < 3; .@i++)
			if (wsv_ch2b & (256 << .@i))
				cloakonnpcself .@f$[.@i];
	} else if (.@m$ == "veledor") {
		setarray .@f$, "Knot of Magic#wsv_c2a", "Knot of Magic#wsv_c2b", "Knot of Magic#wsv_c2c";
		for (.@i = 0; .@i < 3; .@i++)
			if (wsv_ch2b & (1 << .@i))
				cloakonnpcself .@f$[.@i];
	}
	return;
}

// Flame Gold Coins (points: the client has no Flame Gold Coin).
// callfunc "F_WSV_Flame", <coins>;
function	script	F_WSV_Flame	{
	wsv_flame += getarg(0);
	dispbottom "You received " + getarg(0) + " Flame Gold Coin(s) (" + wsv_flame + " in total).";
	return;
}

// Experience for level 225+: a share of the next level, capped.
// callfunc "F_WSV_C2Reward", <percent of the next level>, <coins>;
function	script	F_WSV_C2Reward	{
	getexp2 min(NextBaseExp / 100 * getarg(0), 1500000000), min(NextJobExp / 100 * getarg(0), 750000000);
	if (getarg(1))
		callfunc "F_WSV_Flame", getarg(1);
	return;
}

// 1 when Chapter 1's story is over (this block's step, or its last
// quest done).
function	script	F_WSV_C1Done	{
	return (wsv_ch1 >= 1000 || isbegin_quest(18377) == 2);
}

// A one-time quest or a daily: starts it, or says it's running or
// waiting. Returns 1 when it was started.
// callfunc("F_WSV_C2Daily", "[<npc>]", <quest>, <cooldown>, "<what to do>")
function	script	F_WSV_C2Daily	{
	.@q = getarg(1);
	.@cd = getarg(2);
	if (isbegin_quest(.@q) == 1) {
		mes getarg(0);
		mes "You're still on it: " + getarg(3);
		return 0;
	}
	if (.@cd && callfunc("F_WSV_Waiting", .@cd)) {
		mes getarg(0);
		mes "That's done for today. Come back after 4 AM.";
		return 0;
	}
	if (isbegin_quest(.@q) == 2)
		erasequest .@q;
	setquest .@q;
	questinfo_refresh;
	mes getarg(0);
	mes getarg(3);
	return 1;
}

// Finishes a daily: completes it, starts its cooldown, pays.
// callfunc "F_WSV_C2DailyDone", <quest>, <cooldown>, <percent>, <coins>;
function	script	F_WSV_C2DailyDone	{
	completequest getarg(0);
	erasequest getarg(0);
	if (isbegin_quest(getarg(1)))
		erasequest getarg(1);
	setquest getarg(1);
	callfunc "F_WSV_C2Reward", getarg(2), getarg(3);
	questinfo_refresh;
	return;
}

// The arc quest of a step.
function	script	F_WSV_C2Arc	{
	.@s = getarg(0);
	if (.@s < 100 || .@s >= 1000)
		return 0;
	setarray .@from, 100, 140, 170, 250, 420, 460, 640, 680, 800;
	for (.@i = 8; .@i > 0; .@i--)
		if (.@s >= .@from[.@i])
			break;
	return 64201 + .@i;
}

// Moves the story to step getarg(0): the arc quest changes when the arc
// does, the step's bits clear; getarg(1) percent of a level and
// getarg(2) Flame Gold Coins.
// callfunc "F_WSV_C2Go", <step>{, <percent>, <coins>};
function	script	F_WSV_C2Go	{
	.@o = callfunc("F_WSV_C2Arc", wsv_ch2);
	.@n = callfunc("F_WSV_C2Arc", getarg(0));
	if (.@o != .@n) {
		if (.@o) {
			if (isbegin_quest(.@o) == 0)
				setquest .@o;
			if (isbegin_quest(.@o) == 1)
				completequest .@o;
		}
		if (.@n) {
			if (isbegin_quest(.@n) == 2)
				erasequest .@n;
			if (isbegin_quest(.@n) == 0)
				setquest .@n;
		}
	}
	wsv_ch2 = getarg(0);
	wsv_ch2b = 0;
	if (getarg(1, 0) || getarg(2, 0))
		callfunc "F_WSV_C2Reward", getarg(1, 0), getarg(2, 0);
	questinfo_refresh;
	return;
}

// Pages of dialogue: "[Name]", then lines; "-" turns the page.
// callfunc "F_WSV_C2Lines", "[Name]", "line", "line", "-", "line", ...;
function	script	F_WSV_C2Lines	{
	.@n = getargcount();
	mes getarg(0);
	for (.@i = 1; .@i < .@n; .@i++) {
		if (getarg(.@i) == "-") {
			next;
			mes getarg(0);
		} else
			mes getarg(.@i);
	}
	return;
}

// A story beat: if the story is at step getarg(0), the lines, then the
// next step getarg(1) with getarg(2) coins. Returns 1 if it played.
// callfunc("F_WSV_C2Beat", <step>, <next>, <coins>, "[Name]", lines...)
function	script	F_WSV_C2Beat	{
	if (wsv_ch2 != getarg(0))
		return 0;
	.@n = getargcount();
	mes getarg(3);
	for (.@i = 4; .@i < .@n; .@i++) {
		if (getarg(.@i) == "-") {
			next;
			mes getarg(3);
		} else
			mes getarg(.@i);
	}
	callfunc "F_WSV_C2Go", getarg(1), getarg(2) ? 3 : 0, getarg(2);
	return 1;
}

// Flame Gold Coins (points: the client has no Flame Gold Coin).
// callfunc "F_WSV_Flame", <coins>;
function	script	F_WSV_Flame	{
	wsv_flame += getarg(0);
	dispbottom "You received " + getarg(0) + " Flame Gold Coin(s) (" + wsv_flame + " in total).";
	return;
}

// Experience for level 225+: a share of the next level, capped.
// callfunc "F_WSV_C2Reward", <percent of the next level>, <coins>;
function	script	F_WSV_C2Reward	{
	getexp2 min(NextBaseExp / 100 * getarg(0), 1500000000), min(NextJobExp / 100 * getarg(0), 750000000);
	if (getarg(1))
		callfunc "F_WSV_Flame", getarg(1);
	return;
}

// 1 when Chapter 1's story is over (this block's step, or its last
// quest done).
function	script	F_WSV_C1Done	{
	return (wsv_ch1 >= 1000 || isbegin_quest(18377) == 2);
}

// A one-time quest or a daily: starts it, or says it's running or
// waiting. Returns 1 when it was started.
// callfunc("F_WSV_C2Daily", "[<npc>]", <quest>, <cooldown>, "<what to do>")
function	script	F_WSV_C2Daily	{
	.@q = getarg(1);
	.@cd = getarg(2);
	if (isbegin_quest(.@q) == 1) {
		mes getarg(0);
		mes "You're still on it: " + getarg(3);
		return 0;
	}
	if (.@cd && callfunc("F_WSV_Waiting", .@cd)) {
		mes getarg(0);
		mes "That's done for today. Come back after 4 AM.";
		return 0;
	}
	if (isbegin_quest(.@q) == 2)
		erasequest .@q;
	setquest .@q;
	questinfo_refresh;
	mes getarg(0);
	mes getarg(3);
	return 1;
}

// Finishes a daily: completes it, starts its cooldown, pays.
// callfunc "F_WSV_C2DailyDone", <quest>, <cooldown>, <percent>, <coins>;
function	script	F_WSV_C2DailyDone	{
	completequest getarg(0);
	erasequest getarg(0);
	if (isbegin_quest(getarg(1)))
		erasequest getarg(1);
	setquest getarg(1);
	callfunc "F_WSV_C2Reward", getarg(2), getarg(3);
	questinfo_refresh;
	return;
}

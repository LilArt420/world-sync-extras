function	script	F_WSV_C1Tree	{
	mes "[Small Ash Tree]";
	mes "A young ash tree, no taller than you. Its leaves rustle although there is no wind.";
	if (BaseLevel < 200) {
		mes "For a moment you think you hear singing from between the branches... but it fades.";
		close;
	}
	next;
	if (select("Touch the branches.:Leave it alone.") == 2)
		close;
	mes "As your fingers brush the bark, a tiny light slips out from between the leaves...";
	close2;
	cloakoffnpcself getarg(0);
	specialeffect2 EF_GREENLIGHT;
	end;
}
